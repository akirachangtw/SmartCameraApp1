**UI 設計** (Material3 標準樣式):

@Composable  
fun PhotoInfoBottomSheet(  
    exifInfo: ExifInfo,  
    onDismiss: () \-\> Unit,  
    onCopyInfo: (String) \-\> Unit  
) {  
    ModalBottomSheet(onDismissRequest \= onDismiss) {  
        Column(  
            modifier \= Modifier  
                .fillMaxWidth()  
                .padding(16.dp)  
        ) {  
            // 標題  
            Text("照片資訊", style \= MaterialTheme.typography.headlineSmall)

            Spacer(modifier \= Modifier.height(16.dp))

            // 基本資訊  
            InfoSection(title \= "基本資訊") {  
                InfoRow("檔名", exifInfo.fileName)  
                InfoRow("檔案大小", exifInfo.fileSizeFormatted)  
                InfoRow("解析度", exifInfo.resolution)  
                InfoRow("拍攝時間", exifInfo.dateTime ?: "N/A")  
            }

            // GPS 資訊 (若有)  
            if (exifInfo.hasGpsInfo) {

#### **⚠️ Android 16 合規注意事項**

**16KB 頁面大小對齊要求**:

本專案已在 v2.9.6 達成 Android 16 (16KB 頁面大小) 完全合規，開發過程中需持續維護：

2. **新增依賴檢查流程**（使用 Google 官方工具）:  
   \# 編譯 APK  
   ./gradlew assembleDeviceDebug

   \# 方法 1：使用 Google 官方檢測腳本（推薦）  
   bash docs/test\_results/v2.9.6-Phase2.6/Google-check\_elf\_alignment.sh \\  
        app/build/outputs/apk/device/debug/app-device-debug.apk

   \# 腳本會自動檢查：  
   \# \- APK zip-alignment (需 build-tools 35.0.0-rc3+)  
   \# \- ELF alignment (所有 .so 文件)  
   \# \- 輸出：ALIGNED (綠色) 或 UNALIGNED (紅色)

   \# 方法 2：使用 APK Analyzer (Android Studio)  
   \# Build \> Analyze APK \> 檢查 lib/ 目錄下的 .so 文件

3. **Google 官方檢測腳本說明**:  
   * **路徑**: docs/test\_results/v2.9.6-Phase2.6/Google-check\_elf\_alignment.sh  
   * **功能**:  
     * ZIP 對齊檢查（需 zipalign 工具）  
     * ELF LOAD segment 對齊檢查（使用 objdump）  
     * 支援 APK、APEX、目錄檢查  
   * **對齊標準**:  
     * ✅ ALIGNED: 2\*\*(14-19) 或更高（16KB \= 2^14）  
     * ❌ UNALIGNED: 低於 2^14

   * \[ \] 新增 native library 依賴時，先檢查供應商是否支援 16KB

// data/source/ai/TFLiteSegmentationModel.kt  
class TFLiteSegmentationModel @Inject constructor(  
    @ApplicationContext private val context: Context  
) {  
    private lateinit var interpreter: Interpreter

    fun initialize() {  
        val model \= loadModelFile("models/selfie\_segmenter.tflite")  
        val options \= Interpreter.Options().apply {  
            addDelegate(GpuDelegate())  // GPU 加速  
            setNumThreads(4)  
        }  
        interpreter \= Interpreter(model, options)  
    }

    fun segment(bitmap: Bitmap): SegmentationResult {  
        val startTime \= System.currentTimeMillis()

        // 預處理: Resize to 256x256  
        val inputBitmap \= Bitmap.createScaledBitmap(bitmap, 256, 256, true)  
        val inputBuffer \= convertBitmapToBuffer(inputBitmap)

        // 推理  
        val outputBuffer \= FloatArray(256 \* 256\)  
        interpreter.run(inputBuffer, outputBuffer)

        val inferenceTime \= System.currentTimeMillis() \- startTime

        return SegmentationResult(  
            mask \= outputBuffer,  
            width \= 256,  
            height \= 256,  
            inferenceTime \= inferenceTime  
        )  
    }  
}

precision mediump float;

varying vec2 v\_texCoord;  
uniform sampler2D u\_texture;      // 原始相機幀  
uniform sampler2D u\_mask;         // 人像分割 Mask  
uniform float u\_fillStrength;     // 補光強度 (0.0-1.0)

void main() {  
    vec4 originalColor \= texture2D(u\_texture, v\_texCoord);  
    float maskValue \= texture2D(u\_mask, v\_texCoord).r;

    // 只針對人像區域 (mask \> 0.5) 進行提亮  
    if (maskValue \> 0.5) {  
        // S-Curve 提亮 (保持自然)  
        vec3 brightened \= pow(originalColor.rgb, vec3(1.0 \- u\_fillStrength \* 0.3));

        // 邊緣羽化 (避免生硬邊界)  
        float feather \= smoothstep(0.4, 0.6, maskValue);  
        originalColor.rgb \= mix(originalColor.rgb, brightened, feather);

        // 色溫補償 (避免補光後偏藍)  
        originalColor.r \*= 1.0 \+ u\_fillStrength \* 0.05;  
        originalColor.g \*= 1.0 \+ u\_fillStrength \* 0.03;  
    }

    gl\_FragColor \= originalColor;  
}

#### **3\. 自動補光強度調整**

// domain/usecase/CalculateFillStrengthUseCase.kt  
class CalculateFillStrengthUseCase @Inject constructor() {

    operator fun invoke(  
        lightingCondition: LightingCondition,  
        faceBrightness: Float,  
        backgroundBrightness: Float  
    ): Float {  
        return when (lightingCondition) {  
            is LightingCondition.Backlit \-\> {  
                val severity \= lightingCondition.severity  
                when {  
                    severity \> 2.0 \-\> 1.0f     // 嚴重逆光，最大補光  
                    severity \> 1.5 \-\> 0.7f     // 中度逆光  
                    severity \> 1.2 \-\> 0.4f     // 輕度逆光  
                    else \-\> 0.0f               // 正常光線  
                }  
            }  
            is LightingCondition.LowLight \-\> {  
                // 低光環境，中等補光  
                0.5f  
            }  
            else \-\> 0.0f  // 正常光線，不補光  
        }  
    }  
}

// ImageAnalyzer.kt  
class ImageAnalyzer @Inject constructor(  
    private val mlKitDetector: MLKitDetector,  
    private val tfliteModel: TFLiteSegmentationModel  
) : ImageAnalysis.Analyzer {

    private var frameCount \= 0

    override fun analyze(image: ImageProxy) {  
        frameCount++

        // 每幀執行 ML Kit (快速)  
        val detectionResult \= mlKitDetector.detect(image)

        // 每 3 幀執行 TFLite (較慢)  
        val segmentationResult \= if (frameCount % 3 \== 0\) {  
            val bitmap \= image.toBitmap()  
            tfliteModel.segment(bitmap)  
        } else {  
            null  // 使用上次結果  
        }

        image.close()  
    }  
}

於 v2.11.1 定稿，捨棄原有的「虛線3等分弧線」，採用硬核科幻的純幾何變形設計：

**1\. 搜尋中 (Searching)**

* **動態**: 底座不轉動。讀取光弧從 12 點鐘方向 (0點整方位) 開始，0% 到 100% 讀取的同時透明度 (Alpha) 從 0.0 漸變至 1.0，無限循環掃描。

* **視覺**: 僅顯示內部極細正方形 (Focusing.png)。

**3\. 對焦成功 / 物件追蹤 (Success / Eye-AF)**

* **視覺**: 內部正方形 \+ 外圍四個青色 L 型粗截角 (結合成 Focusing\_Success.png)。  
* **機械咬合動畫 (Mechanical Clamp)**: 觸發 350ms 的 3-phase explicit tween。  
  * Phase 1: Scale 1.0 → 0.75 (150ms, FastOutLinearInEasing) — 咔嗒縮小。  
  * Phase 2: Scale 0.75 → 1.2 (120ms, FastOutSlowInEasing) — 彈回超過原始大小。  
  * Phase 3: Scale 1.2 → 1.0 (80ms, LinearOutSlowInEasing) — 穩定落定。  
* **穩定停留 (Dwell)**: 框停留 1000ms (\~1秒) 後，觸發 500ms 淡化過渡：  
  * 外框 Alpha: 1.0 → 0.4  
  * 線條粗細: 2.5dp → 1.5dp

### **⚠️ ML Kit 視覺神經與效能防線優化 (新增決策)**

**核心目標**: 落實「專業相機化」的 AI 降頻策略與「分級分離」處理，確立邊緣運算 (Edge AI) 效能防線。

**1\. ML Kit Analyzer 影像標籤 (Image Labeling) 分級與分離決策**:

* **Face Detection (極速對焦優先)**：絕對禁止細節標籤 (Landmarks)。強制關閉特徵點 (LANDMARK\_MODE\_NONE) 與輪廓 (CONTOUR\_MODE\_NONE)，解決 Landmark 解析崩潰問題。強制開啟 PERFORMANCE\_MODE\_FAST，確保 CPU 佔用率最低。  
* **Object Detection (場景感知準備)**：需要影像標籤。開啟 .enableClassification() 以獲取物件標籤 (Category/Labels)，為未來的場景識別 (V4.0) 鋪路。開啟 STREAM\_MODE 確保追蹤流暢度。

**2\. 設備分級引擎 (Device Tiering) 實作 (Graceful Degradation)**:

* **高階裝置 (如 Pixel 8\)**：60 FPS 預覽 / 10 FPS AI 採樣 / 開啟完整 UI 特效。  
* **中階裝置 (如 Pixel 7a)**：30 FPS 預覽 / 5-8 FPS AI 採樣 / 關閉 Glassmorphism 模糊。  
* **低階裝置 (Android 7.0+)**：20-30 FPS 預覽 / 僅保留基礎對焦框，關閉所有耗能 AI。

**3\. 非同步視覺欺騙 (Asynchronous Deception)**:

* AI 分析器 (ML Kit) 運行於背景執行緒，UI 渲染 (Compose Spring 動畫) 運行於主執行緒。  
* 利用 OpticalFlowTracker 在 AI 採樣的空窗期進行 60FPS 補幀，達成「低算力高流暢度」的完美錯覺。
