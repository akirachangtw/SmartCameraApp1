**UI 設計** (Material3 標準樣式):
```kotlin
@Composable
fun PhotoInfoBottomSheet(
    exifInfo: ExifInfo,
    onDismiss: () -> Unit,
    onCopyInfo: (String) -> Unit
) {
    ModalBottomSheet(onDismissRequest = onDismiss) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp)
        ) {
            // 標題
            Text("照片資訊", style = MaterialTheme.typography.headlineSmall)

            Spacer(modifier = Modifier.height(16.dp))

            // 基本資訊
            InfoSection(title = "基本資訊") {
                InfoRow("檔名", exifInfo.fileName)
                InfoRow("檔案大小", exifInfo.fileSizeFormatted)
                InfoRow("解析度", exifInfo.resolution)
                InfoRow("拍攝時間", exifInfo.dateTime ?: "N/A")
            }

            // GPS 資訊 (若有)
            if (exifInfo.hasGpsInfo) {
                Spacer(modifier = Modifier.height(8.dp))
                InfoSection(title = "位置資訊") {
                    InfoRow("緯度", String.format("%.6f", exifInfo.latitude))
                    InfoRow("經度", String.format("%.6f", exifInfo.longitude))
                    exifInfo.altitude?.let {
                        InfoRow("海拔", "${it.toInt()} m")
                    }
                }
            }

            // 相機資訊
            Spacer(modifier = Modifier.height(8.dp))
            InfoSection(title = "相機資訊") {
                InfoRow("裝置", "${exifInfo.make} ${exifInfo.model}")
                InfoRow("軟體", exifInfo.software)
                InfoRow("方向", exifInfo.orientationDescription)
            }

            Spacer(modifier = Modifier.height(16.dp))

            // 複製按鈕
            Button(
                onClick = { onCopyInfo(exifInfo.toFormattedString()) },
                modifier = Modifier.fillMaxWidth()
            ) {
                Icon(Icons.Default.ContentCopy, contentDescription = null)
                Spacer(modifier = Modifier.width(8.dp))
                Text("複製資訊")
            }
        }
    }
}
```

```kotlin
class MotionDetector @Inject constructor(
    private val sensorManager: SensorManager
) {
    private val gyroscope = sensorManager.getDefaultSensor(Sensor.TYPE_GYROSCOPE)

    fun detectShake(): Flow<Boolean> = callbackFlow {
        val listener = object : SensorEventListener {
            override fun onSensorChanged(event: SensorEvent) {
                val x = event.values[0]
                val y = event.values[1]
                val z = event.values[2]

                val magnitude = sqrt(x*x + y*y + z*z)

                if (magnitude > SHAKE_THRESHOLD) {
                    trySend(true)  // 晃動偵測
                }
            }
        }
        sensorManager.registerListener(listener, gyroscope, SensorManager.SENSOR_DELAY_GAME)

        awaitClose { sensorManager.unregisterListener(listener) }
    }
```

```kotlin
// data/source/ai/TFLiteSegmentationModel.kt
class TFLiteSegmentationModel @Inject constructor(
    @ApplicationContext private val context: Context
) {
    private lateinit var interpreter: Interpreter

    fun initialize() {
        val model = loadModelFile("models/selfie_segmenter.tflite")
        val options = Interpreter.Options().apply {
            addDelegate(GpuDelegate())  // GPU 加速
            setNumThreads(4)
        }
        interpreter = Interpreter(model, options)
    }

    fun segment(bitmap: Bitmap): SegmentationResult {
        val startTime = System.currentTimeMillis()

        // 預處理: Resize to 256x256
        val inputBitmap = Bitmap.createScaledBitmap(bitmap, 256, 256, true)
        val inputBuffer = convertBitmapToBuffer(inputBitmap)

        // 推理
        val outputBuffer = FloatArray(256 * 256)
        interpreter.run(inputBuffer, outputBuffer)

        val inferenceTime = System.currentTimeMillis() - startTime

        return SegmentationResult(
            mask = outputBuffer,
            width = 256,
            height = 256,
            inferenceTime = inferenceTime
        )
    }
}
```

#### 2. 智能補光 Shader

```glsl
// fragment_smart_fill_light.glsl
precision mediump float;

varying vec2 v_texCoord;
uniform sampler2D u_texture;      // 原始相機幀
uniform sampler2D u_mask;         // 人像分割 Mask
uniform float u_fillStrength;     // 補光強度 (0.0-1.0)

void main() {
    vec4 originalColor = texture2D(u_texture, v_texCoord);
    float maskValue = texture2D(u_mask, v_texCoord).r;

    // 只針對人像區域 (mask > 0.5) 進行提亮
    if (maskValue > 0.5) {
        // S-Curve 提亮 (保持自然)
        vec3 brightened = pow(originalColor.rgb, vec3(1.0 - u_fillStrength * 0.3));

        // 邊緣羽化 (避免生硬邊界)
        float feather = smoothstep(0.4, 0.6, maskValue);
        originalColor.rgb = mix(originalColor.rgb, brightened, feather);

        // 色溫補償 (避免補光後偏藍)
        originalColor.r *= 1.0 + u_fillStrength * 0.05;
        originalColor.g *= 1.0 + u_fillStrength * 0.03;
    }

    gl_FragColor = originalColor;
}
```

```kotlin
// domain/usecase/CalculateFillStrengthUseCase.kt
class CalculateFillStrengthUseCase @Inject constructor() {

    operator fun invoke(
        lightingCondition: LightingCondition,
        faceBrightness: Float,
        backgroundBrightness: Float
    ): Float {
        return when (lightingCondition) {
            is LightingCondition.Backlit -> {
                val severity = lightingCondition.severity
                when {
                    severity > 2.0 -> 1.0f     // 嚴重逆光，最大補光
                    severity > 1.5 -> 0.7f     // 中度逆光
                    severity > 1.2 -> 0.4f     // 輕度逆光
                    else -> 0.0f               // 正常光線
                }
            }
            is LightingCondition.LowLight -> {
                // 低光環境，中等補光
                0.5f
            }
            else -> 0.0f  // 正常光線，不補光
        }
    }
}
```

```kotlin
// ImageAnalyzer.kt
class ImageAnalyzer @Inject constructor(
    private val mlKitDetector: MLKitDetector,
    private val tfliteModel: TFLiteSegmentationModel
) : ImageAnalysis.Analyzer {

    private var frameCount = 0

    override fun analyze(image: ImageProxy) {
        frameCount++

        // 每幀執行 ML Kit (快速)
        val detectionResult = mlKitDetector.detect(image)

        // 每 3 幀執行 TFLite (較慢)
        val segmentationResult = if (frameCount % 3 == 0) {
            val bitmap = image.toBitmap()
            tfliteModel.segment(bitmap)
        } else {
            null  // 使用上次結果
        }

        image.close()
    }
}
```

**不建議實作的功能**:
- ❌ 使用頻率 < 低 且 實作複雜度 = VERY HIGH
- ❌ 需要硬體支援但 < 30% 裝置具備
- ❌ 會導致 FPS 降幅 > 15% 或記憶體增加 > 50%
