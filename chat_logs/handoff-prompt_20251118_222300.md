package local.mylab.smartcameraapp1.data.source.ml

import android.util.Log
import androidx.annotation.OptIn
import androidx.camera.core.ExperimentalGetImage
import androidx.camera.core.ImageProxy
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.face.FaceDetection
import com.google.mlkit.vision.face.FaceDetector
import com.google.mlkit.vision.face.FaceDetectorOptions
import com.google.mlkit.vision.objects.DetectedObject
import com.google.mlkit.vision.objects.ObjectDetection
import com.google.mlkit.vision.objects.ObjectDetector
import com.google.mlkit.vision.objects.defaults.ObjectDetectorOptions
import kotlinx.coroutines.tasks.await
import local.mylab.smartcameraapp1.domain.model.DetectionResult
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class MLKitDetector @Inject constructor() {

    companion object {
        private const val TAG = "MLKitDetector"
    }

    private val faceDetector: FaceDetector by lazy {
        val options = FaceDetectorOptions.Builder()
            .setPerformanceMode(FaceDetectorOptions.PERFORMANCE_MODE_FAST)
            .enableTracking()
            .setMinFaceSize(0.15f)
            .build()
        FaceDetection.getClient(options).also {
            Log.i(TAG, "Face detector initialized with FAST mode, tracking enabled, minFaceSize=0.15f")
        }
    }

    private val objectDetector: ObjectDetector by lazy {
        val options = ObjectDetectorOptions.Builder()
            .setDetectorMode(ObjectDetectorOptions.STREAM_MODE)
            .enableMultipleObjects()
            .enableClassification()
            .build()
        ObjectDetection.getClient(options).also {
            Log.i(TAG, "Object detector initialized with STREAM mode, multiple objects enabled")
        }
    }

    @OptIn(ExperimentalGetImage::class)
    suspend fun detect(imageProxy: ImageProxy): DetectionResult? {
        val mediaImage = imageProxy.image ?: run {
            Log.e(TAG, "Failed to get image from ImageProxy")
            return null
        }

        val inputImage = InputImage.fromMediaImage(
            mediaImage,
            imageProxy.imageInfo.rotationDegrees
        )

        return try {
            val faces = faceDetector.process(inputImage).await()
            val objects = objectDetector.process(inputImage).await()

            DetectionResult(
                faces = faces,
                objects = objects,
                timestamp = System.currentTimeMillis()
            ).also {
                Log.i(TAG, "Detection completed: ${faces.size} faces, ${objects.size} objects")
            }
        } catch (e: Exception) {
            Log.e(TAG, "Detection failed", e)
            null
        }
    }

    fun close() {
        try {
            faceDetector.close()
            objectDetector.close()
            Log.i(TAG, "ML Kit detectors closed")
        } catch (e: Exception) {
            Log.e(TAG, "Failed to close detectors", e)
        }
    }
}

package local.mylab.smartcameraapp1.data.source.ml

import android.util.Log
import androidx.camera.core.ImageAnalysis
import androidx.camera.core.ImageProxy
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.launch
import local.mylab.smartcameraapp1.domain.model.DetectionResult
import java.util.concurrent.Executor
import javax.inject.Inject

class ImageAnalyzer @Inject constructor(
    private val mlKitDetector: MLKitDetector,
    @local.mylab.smartcameraapp1.di.BackgroundExecutor private val executor: Executor
) : ImageAnalysis.Analyzer {

    companion object {
        private const val TAG = "ImageAnalyzer"
        private const val TARGET_FPS = 10
        private const val FRAME_INTERVAL_MS = 1000L / TARGET_FPS
    }

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default)
    private var lastAnalysisTimestamp = 0L
    private var detectionCallback: ((DetectionResult) -> Unit)? = null

    fun setDetectionCallback(callback: (DetectionResult) -> Unit) {
        detectionCallback = callback
        Log.i(TAG, "Detection callback registered")
    }

    override fun analyze(imageProxy: ImageProxy) {
        val currentTime = System.currentTimeMillis()

        if (currentTime - lastAnalysisTimestamp < FRAME_INTERVAL_MS) {
            imageProxy.close()
            return
        }

        lastAnalysisTimestamp = currentTime
        Log.i(TAG, "Frame analysis started at timestamp: $currentTime")

        scope.launch {
            try {
                val result = mlKitDetector.detect(imageProxy)
                result?.let { detectionCallback?.invoke(it) }
            } catch (e: Exception) {
                Log.e(TAG, "Analysis failed", e)
            } finally {
                imageProxy.close()
            }
        }
    }

    fun cleanup() {
        scope.cancel()
        mlKitDetector.close()
        Log.i(TAG, "ImageAnalyzer cleaned up")
    }
}

package local.mylab.smartcameraapp1.presentation.camera

import android.util.Log
import androidx.camera.core.ImageCapture
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.launchIn
import kotlinx.coroutines.flow.onEach
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import local.mylab.smartcameraapp1.domain.model.CaptureResult
import local.mylab.smartcameraapp1.domain.repository.CameraRepository
import javax.inject.Inject

@HiltViewModel
class CameraViewModel @Inject constructor(
    private val cameraRepository: CameraRepository
) : ViewModel() {

    companion object {
        private const val TAG = "CameraViewModel"
    }

    private val _state = MutableStateFlow(CameraState())
    val state: StateFlow<CameraState> = _state.asStateFlow()

    init {
        // 訂閱偵測結果
        observeDetectionResults()

        // ML 偵測需手動啟動（透過 CameraUiEvent.StartAnalysis）
        // 或根據需求自動啟動
        startAnalysis()
    }

    fun getCameraController() = cameraRepository.getCameraController()

    fun onEvent(event: CameraUiEvent) {
        when (event) {
            is CameraUiEvent.CapturePhoto -> capturePhoto()
            is CameraUiEvent.SwitchCamera -> switchCamera()
            is CameraUiEvent.ToggleFlash -> toggleFlash()
            is CameraUiEvent.StartAnalysis -> startAnalysis()
            is CameraUiEvent.StopAnalysis -> stopAnalysis()
            is CameraUiEvent.ShowError -> showError(event.message)
        }
    }

    private fun capturePhoto() {
        viewModelScope.launch {
            _state.update { it.copy(isCapturing = true, error = null) }

            when (val result = cameraRepository.takePicture()) {
                is CaptureResult.Success -> {
                    _state.update {
                        it.copy(
                            isCapturing = false,
                            lastCapturedImageUri = result.uri,
                            error = null
                        )
                    }
                }
                is CaptureResult.Error -> {
                    _state.update {
                        it.copy(
                            isCapturing = false,
                            error = result.exception.message ?: "拍照失敗"
                        )
                    }
                }
            }
        }
    }

    private fun switchCamera() {
        cameraRepository.switchCamera()
        _state.update {
            it.copy(cameraSelector = cameraRepository.getCurrentCameraSelector())
        }
    }

    private fun toggleFlash() {
        val newFlashMode = when (cameraRepository.getFlashMode()) {
            ImageCapture.FLASH_MODE_OFF -> ImageCapture.FLASH_MODE_AUTO
            ImageCapture.FLASH_MODE_AUTO -> ImageCapture.FLASH_MODE_ON
            else -> ImageCapture.FLASH_MODE_OFF
        }
        cameraRepository.setFlashMode(newFlashMode)
        _state.update { it.copy(flashMode = newFlashMode) }
    }

    private fun showError(message: String) {
        _state.update { it.copy(error = message) }
    }

    fun clearError() {
        _state.update { it.copy(error = null) }
    }

    private fun startAnalysis() {
        try {
            cameraRepository.startImageAnalysis()
            _state.update { it.copy(isAnalysisEnabled = true) }
            Log.i(TAG, "ML Kit analysis started")
        } catch (e: Exception) {
            Log.e(TAG, "Failed to start ML Kit analysis", e)
            showError("啟動偵測失敗：${e.message}")
        }
    }

    private fun stopAnalysis() {
        try {
            cameraRepository.stopImageAnalysis()
            _state.update {
                it.copy(
                    isAnalysisEnabled = false,
                    detectionResult = null
                )
            }
            Log.i(TAG, "ML Kit analysis stopped")
        } catch (e: Exception) {
            Log.e(TAG, "Failed to stop ML Kit analysis", e)
        }
    }

    private fun observeDetectionResults() {
        cameraRepository.detectionResults
            .onEach { result ->
                if (result != null) {
                    _state.update { it.copy(detectionResult = result) }
                    Log.i(
                        TAG,
                        "Detection result updated: ${result.faces.size} faces, ${result.objects.size} objects"
                    )
                }
            }
            .launchIn(viewModelScope)
    }
}

package local.mylab.smartcameraapp1.di

import dagger.Module
import dagger.Provides
import dagger.hilt.InstallIn
import dagger.hilt.components.SingletonComponent
import java.util.concurrent.Executor
import java.util.concurrent.Executors
import javax.inject.Qualifier
import javax.inject.Singleton

@Qualifier
@Retention(AnnotationRetention.BINARY)
annotation class BackgroundExecutor

@Module
@InstallIn(SingletonComponent::class)
object AppModule {

    @Provides
    @Singleton
    @BackgroundExecutor
    fun provideBackgroundExecutor(): Executor {
        return Executors.newSingleThreadExecutor { runnable ->
            Thread(runnable, "MLKit-Analysis-Thread").apply {
                priority = Thread.NORM_PRIORITY
            }
        }
    }
}
