## 1. 專案背景 (Context)

SmartCameraApp1 是一個 Android 相機應用，整合 ML Kit 進行即時人臉偵測與物件辨識，並在相機預覽上繪製偵測邊界框。

### 核心問題
v1.3.x 系列版本使用手動座標轉換，導致持續性的邊界框位置錯誤：
- **v1.3.1**: 雙重旋轉 bug
- **v1.3.2**: 維度交換錯誤
- **v1.3.3**: 移除交換但問題仍存
- **v1.3.4**: 過濾使用錯誤座標空間
- **v1.3.5**: 修正過濾但轉換仍有瑕疵

### 解決方案
**v1.4.0 完全重構**：採用 Google 官方 `MlKitAnalyzer` 搭配 `COORDINATE_SYSTEM_VIEW_REFERENCED`，消除所有手動座標轉換邏輯。

```kotlin
package local.mylab.smartcameraapp1.data.repository

import android.content.ContentValues
import android.content.Context
import android.os.Build
import android.provider.MediaStore
import android.util.Log
import androidx.camera.core.CameraSelector
import androidx.camera.core.ImageCapture
import androidx.camera.core.ImageCaptureException
import androidx.camera.mlkit.vision.MlKitAnalyzer
import androidx.camera.view.CameraController.COORDINATE_SYSTEM_VIEW_REFERENCED
import androidx.camera.view.LifecycleCameraController
import androidx.core.content.ContextCompat
import com.google.mlkit.vision.face.FaceDetection
import com.google.mlkit.vision.face.FaceDetector
import com.google.mlkit.vision.face.FaceDetectorOptions
import com.google.mlkit.vision.objects.DetectedObject
import com.google.mlkit.vision.objects.ObjectDetection
import com.google.mlkit.vision.objects.ObjectDetector
import com.google.mlkit.vision.objects.defaults.ObjectDetectorOptions
import dagger.hilt.android.qualifiers.ApplicationContext
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.suspendCancellableCoroutine
import kotlinx.coroutines.withContext
import local.mylab.smartcameraapp1.domain.model.CaptureResult
import local.mylab.smartcameraapp1.domain.model.DetectionResult
import local.mylab.smartcameraapp1.domain.repository.CameraRepository
import local.mylab.smartcameraapp1.util.Constants
import java.text.SimpleDateFormat
import java.util.Locale
import javax.inject.Inject
import javax.inject.Singleton
import kotlin.coroutines.resume

@Singleton
class CameraRepositoryImpl @Inject constructor(
    @ApplicationContext private val context: Context
) : CameraRepository {

    companion object {
        private const val TAG = "CameraRepositoryImpl"

        // Filtering thresholds
        private const val MIN_OBJECT_CONFIDENCE = 0.5f
        private const val MIN_FACE_SIZE = 0.15f
    }

    // Detection results state flow
    private val _detectionResults = MutableStateFlow<DetectionResult?>(null)
    override val detectionResults: StateFlow<DetectionResult?> = _detectionResults.asStateFlow()

    // Analysis started flag
    private var isAnalysisStarted = false

    // ML Kit detectors (lazy initialization)
    private val faceDetector: FaceDetector by lazy {
        val options = FaceDetectorOptions.Builder()
            .setPerformanceMode(FaceDetectorOptions.PERFORMANCE_MODE_FAST)
            .setLandmarkMode(FaceDetectorOptions.LANDMARK_MODE_NONE)
            .setClassificationMode(FaceDetectorOptions.CLASSIFICATION_MODE_NONE)
            .setMinFaceSize(MIN_FACE_SIZE)
            .enableTracking()
            .build()

        FaceDetection.getClient(options).also {
            Log.i(TAG, "Face detector initialized")
        }
    }

    private val objectDetector: ObjectDetector by lazy {
        val options = ObjectDetectorOptions.Builder()
            .setDetectorMode(ObjectDetectorOptions.STREAM_MODE)
            .enableMultipleObjects()
            .enableClassification()
            .build()

        ObjectDetection.getClient(options).also {
            Log.i(TAG, "Object detector initialized")
        }
    }

    private val _cameraController: LifecycleCameraController by lazy {
        LifecycleCameraController(context).apply {
            setEnabledUseCases(
                LifecycleCameraController.IMAGE_CAPTURE or
                LifecycleCameraController.IMAGE_ANALYSIS
            )
            imageCaptureFlashMode = ImageCapture.FLASH_MODE_AUTO
        }
    }

    override fun getCameraController(): LifecycleCameraController = _cameraController

    override suspend fun takePicture(): CaptureResult = withContext(Dispatchers.Main) {
        try {
            Log.d(TAG, "Starting photo capture...")

            suspendCancellableCoroutine<CaptureResult> { continuation ->
                try {
                    val name = SimpleDateFormat(Constants.FILENAME_FORMAT, Locale.getDefault())
                        .format(System.currentTimeMillis())
                    Log.d(TAG, "Photo filename: $name")

                    val contentValues = ContentValues().apply {
                        put(MediaStore.MediaColumns.DISPLAY_NAME, name)
                        put(MediaStore.MediaColumns.MIME_TYPE, "image/jpeg")
                        if (Build.VERSION.SDK_INT > Build.VERSION_CODES.P) {
                            put(MediaStore.Images.Media.RELATIVE_PATH, "Pictures/${Constants.PHOTOS_DIRECTORY}")
                        }
                    }

                    val outputOptions = ImageCapture.OutputFileOptions
                        .Builder(
                            context.contentResolver,
                            MediaStore.Images.Media.EXTERNAL_CONTENT_URI,
                            contentValues
                        )
                        .build()

                    val executor = ContextCompat.getMainExecutor(context)

                    Log.d(TAG, "Initiating camera capture...")
                    _cameraController.takePicture(
                        outputOptions,
                        executor,
                        object : ImageCapture.OnImageSavedCallback {
                            override fun onImageSaved(output: ImageCapture.OutputFileResults) {
                                try {
                                    val savedUri = output.savedUri
                                    if (savedUri != null) {
                                        Log.d(TAG, "Photo saved successfully: $savedUri")
                                        if (continuation.isActive) {
                                            continuation.resume(CaptureResult.Success(savedUri))
                                        }
                                    } else {
                                        Log.e(TAG, "Photo saved but URI is null")
                                        if (continuation.isActive) {
                                            continuation.resume(
                                                CaptureResult.Error(
                                                    Exception("Photo save failed: Unable to get save location")
                                                )
                                            )
                                        }
                                    }
                                } catch (e: Exception) {
                                    Log.e(TAG, "Error in onImageSaved callback", e)
                                    if (continuation.isActive) {
                                        continuation.resume(CaptureResult.Error(e))
                                    }
                                }
                            }

                            override fun onError(exception: ImageCaptureException) {
                                Log.e(TAG, "Photo capture failed: ${exception.message}", exception)
                                if (continuation.isActive) {
                                    continuation.resume(
                                        CaptureResult.Error(
                                            Exception("Photo capture failed: ${exception.message}", exception)
                                        )
                                    )
                                }
                            }
                        }
                    )

                    continuation.invokeOnCancellation {
                        Log.w(TAG, "Photo capture was cancelled")
                    }

                } catch (e: Exception) {
                    Log.e(TAG, "Error setting up photo capture", e)
                    if (continuation.isActive) {
                        continuation.resume(
                            CaptureResult.Error(
                                Exception("Failed to setup photo capture: ${e.message}", e)
                            )
                        )
                    }
                }
            }
        } catch (e: Exception) {
            Log.e(TAG, "Unexpected error in takePicture", e)
            CaptureResult.Error(Exception("Error during photo capture: ${e.message}", e))
        }
    }

    override fun switchCamera() {
        _cameraController.cameraSelector = if (_cameraController.cameraSelector == CameraSelector.DEFAULT_BACK_CAMERA) {
            CameraSelector.DEFAULT_FRONT_CAMERA
        } else {
            CameraSelector.DEFAULT_BACK_CAMERA
        }
    }

    override fun getCurrentCameraSelector(): CameraSelector {
        return _cameraController.cameraSelector
    }

    override fun setFlashMode(flashMode: Int) {
        _cameraController.imageCaptureFlashMode = flashMode
    }

    override fun getFlashMode(): Int {
        return _cameraController.imageCaptureFlashMode
    }

    override fun startImageAnalysis() {
        if (isAnalysisStarted) {
            Log.d(TAG, "Image analysis already started")
            return
        }

        try {
            // Use official MlKitAnalyzer with VIEW_REFERENCED coordinate system
            // This way ML Kit automatically transforms coordinates to PreviewView space
            val mlKitAnalyzer = MlKitAnalyzer(
                listOf(faceDetector, objectDetector),
                COORDINATE_SYSTEM_VIEW_REFERENCED,
                ContextCompat.getMainExecutor(context)
            ) { result: MlKitAnalyzer.Result ->
                // Extract faces and objects from result
                val faces = result.getValue(faceDetector) ?: emptyList()
                val objects = result.getValue(objectDetector) ?: emptyList()

                // Filter low-confidence objects
                val filteredObjects = objects.filter { obj ->
                    val confidence = obj.labels.firstOrNull()?.confidence ?: 0f
                    confidence >= MIN_OBJECT_CONFIDENCE
                }

                // Update detection results
                // Note: Coordinates are already in View space, no need to store image dimensions or rotation
                _detectionResults.value = DetectionResult(
                    faces = faces,
                    objects = filteredObjects,
                    imageWidth = 0,  // No longer needed
                    imageHeight = 0,  // No longer needed
                    rotationDegrees = 0,  // No longer needed
                    timestamp = System.currentTimeMillis()
                )

                Log.d(
                    TAG,
                    "Detection updated: ${faces.size} faces, ${filteredObjects.size} objects (${objects.size - filteredObjects.size} filtered)"
                )
            }

            // Attach MlKitAnalyzer to camera controller
            _cameraController.setImageAnalysisAnalyzer(
                ContextCompat.getMainExecutor(context),
                mlKitAnalyzer
            )

            isAnalysisStarted = true
            Log.i(TAG, "Image analysis started with MlKitAnalyzer (VIEW_REFERENCED coordinates)")
        } catch (e: Exception) {
            Log.e(TAG, "Failed to start image analysis", e)
        }
    }

    override fun stopImageAnalysis() {
        if (!isAnalysisStarted) {
            Log.d(TAG, "Image analysis not started")
            return
        }

        try {
            _cameraController.clearImageAnalysisAnalyzer()
            _detectionResults.value = null
            isAnalysisStarted = false
            Log.i(TAG, "Image analysis stopped")
        } catch (e: Exception) {
            Log.e(TAG, "Failed to stop image analysis", e)
        }
    }
}
```

```kotlin
/**
 * Detection overlay composable
 *
 * Renders bounding boxes for detected faces and objects on camera preview
 *
 * IMPORTANT: With v1.4.0, coordinates from MlKitAnalyzer are already in VIEW space,
 * so we can draw them directly without any transformation!
 *
 * @param detectionResult ML Kit detection result (coordinates already in View space)
 * @param modifier Modifier for the canvas
 */
@Composable
fun DetectionOverlay(
    detectionResult: DetectionResult?,
    modifier: Modifier = Modifier
) {
    // Animated bounding boxes for smooth transitions
    val animatedFaceBoxes = remember {
        mutableMapOf<Int, Animatable<Rect, AnimationVector4D>>()
    }
    val animatedObjectBoxes = remember {
        mutableMapOf<Int, Animatable<Rect, AnimationVector4D>>()
    }

    // Update animations when detection results change
    LaunchedEffect(detectionResult) {
        detectionResult?.let { result ->
            // Animate face boxes
            // IMPORTANT: Coordinates are already in View space, no transformation needed!
            result.faces.forEachIndexed { index, face ->
                val trackingId = face.trackingId ?: index
                val screenRect = face.boundingBox.toComposeRect()

                val animatable = animatedFaceBoxes.getOrPut(trackingId) {
                    Animatable(screenRect, Rect.VectorConverter)
                }

                animatable.animateTo(
                    targetValue = screenRect,
                    animationSpec = tween(durationMillis = 150)
                )
            }

            // Animate object boxes
            // IMPORTANT: Coordinates are already in View space, no transformation needed!
            result.objects.forEachIndexed { index, obj ->
                val trackingId = obj.trackingId ?: index
                val screenRect = obj.boundingBox.toComposeRect()

                val animatable = animatedObjectBoxes.getOrPut(trackingId) {
                    Animatable(screenRect, Rect.VectorConverter)
                }

                animatable.animateTo(
                    targetValue = screenRect,
                    animationSpec = tween(durationMillis = 150)
                )
            }

            // Remove stale animations
            val currentFaceIds = result.faces.mapIndexed { index, face ->
                face.trackingId ?: index
            }.toSet()
            animatedFaceBoxes.keys.retainAll(currentFaceIds)

            val currentObjectIds = result.objects.mapIndexed { index, obj ->
                obj.trackingId ?: index
            }.toSet()
            animatedObjectBoxes.keys.retainAll(currentObjectIds)
        }
    }

    Canvas(modifier = modifier.fillMaxSize()) {
        detectionResult?.let { result ->
            // Draw face bounding boxes
            result.faces.forEachIndexed { index, face ->
                val trackingId = face.trackingId ?: index
                animatedFaceBoxes[trackingId]?.value?.let { rect ->
                    drawFaceBoundingBox(
                        rect = rect,
                        trackingId = trackingId,
                        face = face
                    )
                }
            }

            // Draw object bounding boxes
            result.objects.forEachIndexed { index, obj ->
                val trackingId = obj.trackingId ?: index
                animatedObjectBoxes[trackingId]?.value?.let { rect ->
                    drawObjectBoundingBox(
                        rect = rect,
                        trackingId = trackingId,
                        detectedObject = obj
                    )
                }
            }
        }
    }
}

/**
 * Draw face bounding box with tracking ID
 */
private fun DrawScope.drawFaceBoundingBox(
    rect: Rect,
    trackingId: Int,
    face: Face
) {
    val strokeWidth = 4f
    val color = Color(0xFF4CAF50) // Green

    // Draw bounding box
    drawRect(
        color = color,
        topLeft = rect.topLeft,
        size = Size(rect.width, rect.height),
        style = Stroke(width = strokeWidth)
    )

    // Draw tracking ID label
    val labelText = "ID: $trackingId"
    val textSize = 14.sp.toPx()
    val textPaint = Paint().asFrameworkPaint().apply {
        this.color = color.toArgb()
        this.textSize = textSize
        isAntiAlias = true
        isFakeBoldText = true
    }

    // Background for label
    val textBounds = android.graphics.Rect()
    textPaint.getTextBounds(labelText, 0, labelText.length, textBounds)
    val labelWidth = textBounds.width() + 16f
    val labelHeight = textBounds.height() + 12f

    // Draw label background
    drawRect(
        color = color,
        topLeft = Offset(rect.left, rect.top - labelHeight),
        size = Size(labelWidth, labelHeight)
    )

    // Draw label text
    drawContext.canvas.nativeCanvas.drawText(
        labelText,
        rect.left + 8f,
        rect.top - 6f,
        textPaint.apply {
            this.color = Color.White.toArgb()
        }
    )
}

/**
 * Draw object bounding box with label and confidence
 */
private fun DrawScope.drawObjectBoundingBox(
    rect: Rect,
    trackingId: Int,
    detectedObject: DetectedObject
) {
    val strokeWidth = 4f
    val color = Color(0xFF2196F3) // Blue

    // Draw bounding box
    drawRect(
        color = color,
        topLeft = rect.topLeft,
        size = Size(rect.width, rect.height),
        style = Stroke(width = strokeWidth)
    )

    // Get object label and confidence
    val label = detectedObject.labels.firstOrNull()
    val labelText = if (label != null) {
        val confidence = (label.confidence * 100).toInt()
        "${label.text} $confidence%"
    } else {
        "Object #$trackingId"
    }

    val textSize = 14.sp.toPx()
    val textPaint = Paint().asFrameworkPaint().apply {
        this.color = color.toArgb()
        this.textSize = textSize
        isAntiAlias = true
        isFakeBoldText = true
    }

    // Background for label
    val textBounds = android.graphics.Rect()
    textPaint.getTextBounds(labelText, 0, labelText.length, textBounds)
    val labelWidth = textBounds.width() + 16f
    val labelHeight = textBounds.height() + 12f

    // Draw label background
    drawRect(
        color = color,
        topLeft = Offset(rect.left, rect.top - labelHeight),
        size = Size(labelWidth, labelHeight)
    )

    // Draw label text
    drawContext.canvas.nativeCanvas.drawText(
        labelText,
        rect.left + 8f,
        rect.top - 6f,
        textPaint.apply {
            this.color = Color.White.toArgb()
        }
    )
}

/**
 * Extension function to convert android.graphics.Rect to Compose Rect
 *
 * With MlKitAnalyzer using COORDINATE_SYSTEM_VIEW_REFERENCED, the bounding boxes
 * are already in PreviewView coordinates, so we just need a simple type conversion!
 */
private fun android.graphics.Rect.toComposeRect(): Rect {
    return Rect(
        left = left.toFloat(),
        top = top.toFloat(),
        right = right.toFloat(),
        bottom = bottom.toFloat()
    )
}
```

```kotlin
package local.mylab.smartcameraapp1.presentation.camera.components

import androidx.camera.view.LifecycleCameraController
import androidx.camera.view.PreviewView
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalLifecycleOwner
import androidx.compose.ui.viewinterop.AndroidView

/**
 * Camera Preview composable
 *
 * v1.4.0: Simplified - no longer needs size tracking callback since
 * MlKitAnalyzer with VIEW_REFERENCED handles coordinate transformation automatically
 */
@Composable
fun CameraPreview(
    controller: LifecycleCameraController,
    modifier: Modifier = Modifier
) {
    val lifecycleOwner = LocalLifecycleOwner.current

    AndroidView(
        factory = { context ->
            PreviewView(context).apply {
                // Set scale type to FIT_CENTER to maintain aspect ratio
                // This ensures MlKitAnalyzer's VIEW_REFERENCED coordinates work correctly
                scaleType = PreviewView.ScaleType.FIT_CENTER

                this.controller = controller
                controller.bindToLifecycle(lifecycleOwner)
            }
        },
        modifier = modifier
    )
}
```

### 🔧 如遇到問題

**問題 1: Gradle 建置失敗**
- 檢查 `libs.versions.toml` 是否包含 `androidx-camera-mlkit-vision` 條目
- 檢查 `app/build.gradle.kts` 是否有 `implementation(libs.androidx.camera.mlkit.vision)`
- 執行 `./gradlew clean` 後重新建置

**問題 2: 檔案被 linter 恢復**
- 系統可能會自動格式化或恢復檔案
- 重新檢查關鍵檔案（CameraRepositoryImpl.kt、DetectionOverlay.kt）
- 確認是否包含 `COORDINATE_SYSTEM_VIEW_REFERENCED`
- 如需要，從 commit 3c816e5 重新 checkout

**問題 3: 座標仍然不正確**
- 驗證 PreviewView 的 scaleType 為 `FIT_CENTER`
- 確認 DetectionOverlay 沒有進行任何座標轉換
- 檢查是否正確使用 `toComposeRect()` 而非 `transformRect()`
