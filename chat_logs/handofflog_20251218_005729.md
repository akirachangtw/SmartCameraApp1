## 1. 專案背景 (Context)

### 專案概述
Smart Camera App 是一個 Android 攝影助手應用程式，使用 ML Kit 進行即時臉部偵測和構圖分析，提供專業攝影建議給使用者。

### 當前工作重點
我們正在實作「**Camera Input 抽象層**」，目的是將 CameraX 即時畫面與測試圖片來源統一成同一介面：
- **LiveCameraSource** - 實機 CameraX 即時預覽
- **StaticImageSource** - 模擬器/測試模式（循環播放測試圖片）

這個抽象層讓 UI 和構圖引擎完全解耦於 CameraX，使得模擬器開發和單元測試更加容易。

```kotlin
    Timber.d("LiveCameraSource: Taking picture with real camera")

    _cameraController.takePicture(
        ContextCompat.getMainExecutor(context),
        object : ImageCapture.OnImageCapturedCallback() {
            override fun onCaptureSuccess(image: ImageProxy) {
                Timber.d("LiveCameraSource: Image captured successfully (${image.width}x${image.height})")
                try {
                    val bitmap = image.toBitmap()
                    Timber.i("LiveCameraSource: Bitmap created from captured image: ${bitmap.width}x${bitmap.height}")
                    image.close()

                } catch (e: Exception) {
                    Timber.e(e, "LiveCameraSource: Failed to convert captured ImageProxy to Bitmap")
                    image.close()
                    continuation.resume(Result.failure(e))
                }
            }

            override fun onError(exception: ImageCaptureException) {
                Timber.e(exception, "LiveCameraSource: Failed to capture image")
                continuation.resume(
                    Result.failure(
                        Exception("Failed to capture image: ${exception.message}", exception)
                    )
                )
            }
        }
    )

    continuation.invokeOnCancellation {
        Timber.w("LiveCameraSource: takePicture() was cancelled")
    }
}
```

```kotlin
@SuppressLint("UnsafeOptInUsageError")
private fun ImageProxy.toBitmap(): Bitmap {
    val yPlane = planes[0]
    val uPlane = planes[1]
    val vPlane = planes[2]

    val yBuffer = yPlane.buffer
    val uBuffer = uPlane.buffer
    val vBuffer = vPlane.buffer

    // Reset buffer positions
    yBuffer.rewind()
    uBuffer.rewind()
    vBuffer.rewind()

    val yRowStride = yPlane.rowStride
    val uvRowStride = uPlane.rowStride
    val uvPixelStride = uPlane.pixelStride

    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
    val pixels = IntArray(width * height)

    var pixelIndex = 0
    for (y in 0 until height) {
        val yRowOffset = y * yRowStride
        val uvRowOffset = (y / 2) * uvRowStride

        for (x in 0 until width) {
            // Get Y value
            val yValue = (yBuffer.get(yRowOffset + x).toInt() and 0xFF)

            // Get UV values (subsampled 2x2)
            val uvPixelIndex = uvRowOffset + (x / 2) * uvPixelStride
            val uValue = (uBuffer.get(uvPixelIndex).toInt() and 0xFF) - 128
            val vValue = (vBuffer.get(uvPixelIndex).toInt() and 0xFF) - 128

            // YUV → RGB conversion (ITU-R BT.601)
            val r = (yValue + 1.402f * vValue).toInt().coerceIn(0, 255)
            val g = (yValue - 0.344f * uValue - 0.714f * vValue).toInt().coerceIn(0, 255)
            val b = (yValue + 1.772f * uValue).toInt().coerceIn(0, 255)

            // Pack ARGB into int (0xAARRGGBB)
            pixels[pixelIndex++] = (0xFF shl 24) or (r shl 16) or (g shl 8) or b
        }
    }

    bitmap.setPixels(pixels, 0, width, 0, 0, width, height)
    return bitmap
}
```

#### 4.1 CameraInputSource.kt（介面定義）

```kotlin
package com.akirachang.ai.photographer.data.source.camera

import android.graphics.Bitmap
import kotlinx.coroutines.flow.Flow

/**
 * Camera Input Source Abstraction
 *
 * Provides a unified interface for camera preview and capture operations,
 * abstracting away the underlying implementation (CameraX for real devices,
 * test images for emulators).
 *
 * Implementations:
 * - LiveCameraSource: Real camera using CameraX
 * - StaticImageSource: Test images for emulator/testing
 */
interface CameraInputSource {

    /**
     * Get a continuous stream of preview frames as Bitmaps
     *
     * @return Flow of Bitmap frames from the camera or test source
     */
    fun getPreviewStream(): Flow<Bitmap>

    /**
     * Capture a single high-quality photo
     *
     * @return Result containing the captured Bitmap or error
     */
    suspend fun takePicture(): Result<Bitmap>

    /**
     * Stop the preview stream and release resources
     */
    fun stopPreview()

    /**
     * Set camera zoom ratio
     *
     * @param ratio Zoom ratio (1.0 = no zoom, 2.0 = 2x zoom, etc.)
     * @return Result indicating success or failure
     */
    suspend fun setZoomRatio(ratio: Float): Result<Unit>
}
```

#### 4.2 StaticImageSource.kt（模擬器測試實作）

**檔案路徑：** `app/src/main/java/com/akirachang/ai/photographer/data/source/camera/StaticImageSource.kt`

```kotlin
package com.akirachang.ai.photographer.data.source.camera

import android.graphics.Bitmap
import android.util.Log
import com.akirachang.ai.photographer.data.local.TestImageProvider
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.flow
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Static Image Source for Emulator Testing
 *
 * Provides a camera-like experience by cycling through test images
 * at a fixed interval (2 seconds per image).
 *
 * Used when running on emulators or when EMULATOR_MODE is enabled.
 */
@Singleton
class StaticImageSource @Inject constructor(
    private val testImageProvider: TestImageProvider
) : CameraInputSource {

    companion object {
        private const val TAG = "StaticImageSource"
        private const val FRAME_INTERVAL_MS = 2000L  // 2 seconds between frames
    }

    private var isActive = false
    private var frameCount = 0

    override fun getPreviewStream(): Flow<Bitmap> = flow {
        isActive = true
        frameCount = 0

        Log.i(TAG, "Starting preview stream (test image cycling)")

        while (isActive) {
            frameCount++
            Log.d(TAG, "Fetching next test image (frame #$frameCount)...")

            val bitmap = testImageProvider.getNextTestImage()
            if (bitmap != null) {
                Log.i(TAG, "Emitting test image: ${bitmap.width}x${bitmap.height} (frame #$frameCount)")
                emit(bitmap)
            } else {
                Log.e(TAG, "Failed to load test image for frame #$frameCount")
            }

            Log.d(TAG, "Waiting $FRAME_INTERVAL_MS ms before next frame...")
            delay(FRAME_INTERVAL_MS)
        }

        Log.i(TAG, "Preview stream stopped after $frameCount frames")
    }

    override suspend fun takePicture(): Result<Bitmap> {
        Log.i(TAG, "takePicture: Loading portrait_good_composition.jpg")

        val bitmap = testImageProvider.getTestImage("portrait_good_composition.jpg")
        return if (bitmap != null) {
            Log.i(TAG, "Successfully captured test image: ${bitmap.width}x${bitmap.height}")
            Result.success(bitmap)
        } else {
            Log.e(TAG, "Failed to load test image for takePicture")
            Result.failure(Exception("Failed to load test image"))
        }
    }

    override fun stopPreview() {
        Log.i(TAG, "Stopping preview stream")
        isActive = false
    }

    override suspend fun setZoomRatio(ratio: Float): Result<Unit> {
        Log.d(TAG, "setZoomRatio: $ratio (no-op for static images)")
        return Result.success(Unit)
    }
}
```

#### 4.3 TestImageProvider.kt（測試圖片載入器）

**檔案路徑：** `app/src/main/java/com/akirachang/ai/photographer/data/local/TestImageProvider.kt`

```kotlin
package com.akirachang.ai.photographer.data.local

import android.content.Context
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.util.Log
import dagger.hilt.android.qualifiers.ApplicationContext
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import timber.log.Timber
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Test Image Provider for Emulator Development
 *
 * Loads test images from assets for use in emulator builds.
 * Not used on real devices.
 *
 * Test images are located in: app/src/debug/assets/test_images/
 */
@Singleton
class TestImageProvider @Inject constructor(
    @ApplicationContext private val context: Context
) {
    companion object {
        private const val TAG = "TestImageProvider"
    }

    private val testImages = listOf(
        "portrait_good_composition.jpg",
        "portrait_bad_headroom.jpg",
        "portrait_off_center.jpg",
        "landscape_symmetry.jpg",
        "landscape_rule_of_thirds.jpg"
    )

    private var currentIndex = 0

    /**
     * Load the next test image in the rotation
     *
     * @return Bitmap of the next test image, or fallback image if loading fails
     */
    suspend fun getNextTestImage(): Bitmap? = withContext(Dispatchers.IO) {
        val imageName = testImages[currentIndex]
        Log.i(TAG, "Loading test image [$currentIndex]: $imageName")

        try {
            Log.d(TAG, "Attempting to open: test_images/$imageName")

            val bitmap = context.assets.open("test_images/$imageName").use { inputStream ->
                BitmapFactory.decodeStream(inputStream)
            }

            if (bitmap != null) {
                Log.i(TAG, "Successfully loaded: $imageName (${bitmap.width}x${bitmap.height})")
                currentIndex = (currentIndex + 1) % testImages.size
                bitmap
            } else {
                Log.e(TAG, "BitmapFactory.decodeStream() returned null for: $imageName")
                Timber.e("Failed to decode bitmap: $imageName")
                currentIndex = (currentIndex + 1) % testImages.size
                createFallbackTestImage()
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed to load test image: $imageName", e)
            Timber.e(e, "Failed to load test image")
            currentIndex = (currentIndex + 1) % testImages.size
            createFallbackTestImage()
        }
    }

    /**
     * Load a specific test image by name
     *
     * @param name Filename of the test image (e.g., "portrait_good_composition.jpg")
     * @return Bitmap of the requested image, or fallback image if loading fails
     */
    suspend fun getTestImage(name: String): Bitmap? = withContext(Dispatchers.IO) {
        Log.i(TAG, "Loading specific test image: $name")

        try {
            Log.d(TAG, "Attempting to open: test_images/$name")

            val bitmap = context.assets.open("test_images/$name").use { inputStream ->
                BitmapFactory.decodeStream(inputStream)
            }

            if (bitmap != null) {
                Log.i(TAG, "Successfully loaded specific image: $name (${bitmap.width}x${bitmap.height})")
                bitmap
            } else {
                Log.e(TAG, "BitmapFactory.decodeStream() returned null for: $name")
                Timber.e("Failed to decode bitmap: $name")
                createFallbackTestImage()
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed to load test image: $name", e)
            Timber.e(e, "Failed to load test image: $name")
            createFallbackTestImage()
        }
    }

    /**
     * Get list of available test image filenames
     *
     * @return List of test image filenames
     */
    fun getAvailableTestImages(): List<String> = testImages

    /**
     * Create a fallback test image when actual test images fail to load
     *
     * Generates a simple gray background with a white rectangle in the center.
     *
     * @return Fallback Bitmap (1920x1080)
     */
    private fun createFallbackTestImage(): Bitmap {
        val width = 1920
        val height = 1080
        Log.w(TAG, "Creating fallback test image: ${width}x${height}")

        val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
```

```kotlin
@Provides
@Singleton
fun provideCameraInputSource(
    staticImageSource: StaticImageSource,
    liveCameraSource: LiveCameraSource
): CameraInputSource {
    val isEmulator = isRunningOnEmulator()
    val useEmulatorMode = BuildConfig.EMULATOR_MODE || BuildConfig.IS_EMULATOR_BUILD

    Log.d(TAG, "Emulator detection: isEmulator=$isEmulator, useEmulatorMode=$useEmulatorMode")

    return when {
        useEmulatorMode || isEmulator -> {
            Log.i(TAG, "Using StaticImageSource (emulator/test mode)")
            staticImageSource
        }
        else -> {
            Log.i(TAG, "Using LiveCameraSource (real device)")
            liveCameraSource
        }
    }
}

private fun isRunningOnEmulator(): Boolean {
    return (Build.FINGERPRINT.startsWith("generic")
            || Build.FINGERPRINT.startsWith("unknown")
            || Build.MODEL.contains("google_sdk")
            || Build.MODEL.contains("Emulator")
            || Build.MODEL.contains("Android SDK built for x86")
            || Build.MANUFACTURER.contains("Genymotion")
            || Build.PRODUCT.contains("sdk_google")
            || Build.PRODUCT.contains("google_sdk")
            || Build.PRODUCT.contains("sdk")
            || Build.PRODUCT.contains("sdk_x86")
            || Build.PRODUCT.contains("vbox86p")
            || Build.HARDWARE.contains("goldfish")
            || Build.HARDWARE.contains("ranchu"))
}
```
