### 4.1 核心程式碼 - CameraModule.kt（完整版本）

**檔案路徑**: `app/src/main/java/local/mylab/smartcameraapp1/di/CameraModule.kt`

```kotlin
package local.mylab.smartcameraapp1.di

import android.os.Build
import dagger.Module
import dagger.Provides
import dagger.hilt.InstallIn
import dagger.hilt.components.SingletonComponent
import local.mylab.smartcameraapp1.BuildConfig
import local.mylab.smartcameraapp1.data.source.camera.CameraDataSource
import local.mylab.smartcameraapp1.data.source.camera.FakeCameraDataSource
import local.mylab.smartcameraapp1.data.source.camera.RealCameraDataSource
import timber.log.Timber
import javax.inject.Singleton

/**
 * Hilt DI module for Camera-related dependencies
 *
 * Automatically detects if running on emulator and provides appropriate
 * CameraDataSource implementation (FakeCameraDataSource or RealCameraDataSource)
 */
@Module
@InstallIn(SingletonComponent::class)
object CameraModule {

    /**
     * Provides CameraDataSource implementation based on runtime environment
     *
     * Priority:
     * 1. BuildConfig.EMULATOR_MODE (Gradle flavor setting)
     * 2. BuildConfig.IS_EMULATOR_BUILD (Gradle flavor setting)
     * 3. isRunningOnEmulator() (Runtime detection)
     *
     * @param fakeCameraDataSource Fake implementation using test images
     * @param realCameraDataSource Real implementation using CameraX
     * @return CameraDataSource implementation
     */
    @Provides
    @Singleton
    fun provideCameraDataSource(
        fakeCameraDataSource: FakeCameraDataSource,
        realCameraDataSource: RealCameraDataSource
    ): CameraDataSource {
        val isEmulator = isRunningOnEmulator()
        val useEmulatorMode = BuildConfig.EMULATOR_MODE || BuildConfig.IS_EMULATOR_BUILD

        return when {
            useEmulatorMode || isEmulator -> {
                Timber.d("🎮 Using FakeCameraDataSource (Emulator Mode)")
                Timber.d("  - BuildConfig.EMULATOR_MODE: ${BuildConfig.EMULATOR_MODE}")
                Timber.d("  - BuildConfig.IS_EMULATOR_BUILD: ${BuildConfig.IS_EMULATOR_BUILD}")
                Timber.d("  - Runtime emulator detection: $isEmulator")
                fakeCameraDataSource
            }
            else -> {
                Timber.d("📱 Using RealCameraDataSource (Device Mode)")
                realCameraDataSource
            }
        }
    }

    /**
     * Detects if the app is running on an emulator
     *
     * Uses multiple heuristics to detect emulator:
     * - Build.FINGERPRINT
     * - Build.MODEL
     * - Build.MANUFACTURER
     * - Build.BRAND
     * - Build.DEVICE
     * - Build.PRODUCT
     * - Build.HARDWARE
     *
     * @return true if running on emulator, false otherwise
     */
    private fun isRunningOnEmulator(): Boolean {
        return (Build.FINGERPRINT.startsWith("generic")
                || Build.FINGERPRINT.startsWith("unknown")
                || Build.MODEL.contains("google_sdk")
                || Build.MODEL.contains("Emulator")
                || Build.MODEL.contains("Android SDK built for x86")
                || Build.MANUFACTURER.contains("Genymotion")
                || (Build.BRAND.startsWith("generic") && Build.DEVICE.startsWith("generic"))
                || "google_sdk" == Build.PRODUCT
                || Build.PRODUCT.contains("sdk_gphone")
                || Build.PRODUCT.contains("vbox86p")
                || Build.HARDWARE.contains("goldfish")
                || Build.HARDWARE.contains("ranchu"))
    }
}
```

### 4.2 核心程式碼 - TestImageProvider.kt（完整版本）

**檔案路徑**: `app/src/main/java/local/mylab/smartcameraapp1/data/local/TestImageProvider.kt`

```kotlin
package local.mylab.smartcameraapp1.data.local

import android.content.Context
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Paint
import dagger.hilt.android.qualifiers.ApplicationContext
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import timber.log.Timber
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Provides test images for emulator development and testing
 *
 * Features:
 * - Loads test images from assets/test_images/
 * - Cycles through multiple test images
 * - Generates fallback images when assets are missing
 * - Thread-safe image loading with coroutines
 */
@Singleton
class TestImageProvider @Inject constructor(
    @ApplicationContext private val context: Context
) {
    companion object {
        private const val TAG = "TestImageProvider"

        // Test image file names (stored in app/src/debug/assets/test_images/)
        private val TEST_IMAGE_FILES = listOf(
            "portrait_good_composition.jpg",
            "portrait_bad_headroom.jpg",
            "portrait_off_center.jpg",
            "landscape_symmetry.jpg",
            "landscape_rule_of_thirds.jpg"
        )

        // Fallback image dimensions
        private const val FALLBACK_WIDTH = 1920
        private const val FALLBACK_HEIGHT = 1080
        private const val SUBJECT_SIZE = 400
    }

    private var currentIndex = 0

    /**
     * Gets the next test image in the cycle
     *
     * Attempts to load from assets first, falls back to generated image if loading fails.
     * Automatically cycles through TEST_IMAGE_FILES list.
     *
     * @return Bitmap of the test image, or null if both loading and generation fail
     */
    suspend fun getNextTestImage(): Bitmap? = withContext(Dispatchers.IO) {
        try {
            val imageName = TEST_IMAGE_FILES[currentIndex]
            currentIndex = (currentIndex + 1) % TEST_IMAGE_FILES.size

            Timber.d("Loading test image: $imageName (index: ${currentIndex - 1})")

            context.assets.open("test_images/$imageName").use { inputStream ->
                val bitmap = BitmapFactory.decodeStream(inputStream)
                if (bitmap != null) {
                    Timber.d("Successfully loaded test image: $imageName (${bitmap.width}x${bitmap.height})")
                    bitmap
                } else {
                    Timber.w("Failed to decode test image: $imageName, using fallback")
                    createFallbackTestImage()
                }
            }
        } catch (e: Exception) {
            Timber.e(e, "Failed to load test image from assets, generating fallback")
            createFallbackTestImage()
        }
    }

    /**
     * Creates a fallback test image when asset loading fails
     *
     * Generates a simple image with:
     * - Gray background (simulating environment)
     * - White rectangle in center (simulating subject/face)
     *
     * Dimensions: 1920x1080 (Full HD)
     * Subject size: 400x400 (centered)
     *
     * @return Generated bitmap, or null if creation fails
     */
    private fun createFallbackTestImage(): Bitmap? {
        return try {
            Timber.d("Generating fallback test image (${FALLBACK_WIDTH}x${FALLBACK_HEIGHT})")

            val bitmap = Bitmap.createBitmap(
                FALLBACK_WIDTH,
                FALLBACK_HEIGHT,
                Bitmap.Config.ARGB_8888
            )

            val canvas = Canvas(bitmap)

            // Draw gray background
            canvas.drawColor(Color.GRAY)

            // Draw white rectangle in center (simulating subject)
            val paint = Paint().apply {
                color = Color.WHITE
                style = Paint.Style.FILL
            }

            val left = (FALLBACK_WIDTH - SUBJECT_SIZE) / 2f
            val top = (FALLBACK_HEIGHT - SUBJECT_SIZE) / 2f
            val right = left + SUBJECT_SIZE
            val bottom = top + SUBJECT_SIZE

            canvas.drawRect(left, top, right, bottom, paint)

            Timber.d("Successfully generated fallback test image")
            bitmap
        } catch (e: Exception) {
            Timber.e(e, "Failed to generate fallback test image")
            null
        }
    }

    /**
     * Resets the image cycle to the first image
     */
    fun resetCycle() {
        currentIndex = 0
        Timber.d("Reset test image cycle to index 0")
    }
}
```
