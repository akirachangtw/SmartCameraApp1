package com.akirachang.ai.photographer.presentation.camera

import android.util.Log

sealed class FocusDecision {
    object Allow : FocusDecision()
    object SyntheticLock : FocusDecision()  // fixed-focus bypass
    object Reject : FocusDecision()          // front camera manual tap
}

object FocusOrchestrator {

    private const val TAG = "FocusOrchestrator"

    fun decide(
        isFixedFocusLens: Boolean,
        isFrontCamera: Boolean,
        isUserTap: Boolean
    ): FocusDecision {
        return when {
            isFixedFocusLens -> {
                Log.i(TAG, "[v2.12.0-14] FocusOrchestrator.decide: SyntheticLock (isFixedFocus=true)")
                FocusDecision.SyntheticLock
            }
            isFrontCamera && isUserTap -> {
                Log.i(TAG, "[v2.12.0-14] FocusOrchestrator.decide: Reject (front camera manual tap)")
                FocusDecision.Reject
            }
            else -> FocusDecision.Allow
        }
    }
}

private fun extractEyePositionsFromFaces(faces: List<Face>): List<EyePosition> {
    return faces.mapNotNull { face ->
        val leftEye = face.getLandmark(FaceLandmark.LEFT_EYE)?.position
        val rightEye = face.getLandmark(FaceLandmark.RIGHT_EYE)?.position
        val nearestEye = calculateNearestEyePosition(leftEye, rightEye, face.headEulerAngleY)
        if (leftEye != null || rightEye != null) {
            EyePosition(
                leftEye = leftEye,
                rightEye = rightEye,
                nearestEye = nearestEye,
                faceId = face.trackingId ?: -1
            ).also {
                Log.d(TAG, "Eye-AF: Face ${face.trackingId} - Left: $leftEye, Right: $rightEye, Nearest: $nearestEye (Yaw: ${face.headEulerAngleY}°)")
            }
        } else {
            null
        }
    }
}

private fun calculateNearestEyePosition(
    leftEye: PointF?,
    rightEye: PointF?,
    headEulerAngleY: Float
): PointF? {
    return when {
        headEulerAngleY > 10f -> rightEye
        headEulerAngleY < -10f -> leftEye
        else -> {
            if (leftEye != null && rightEye != null) {
                if (leftEye.y < rightEye.y) leftEye else rightEye
            } else {
                leftEye ?: rightEye
            }
        }
    }
}
