```kotlin
@Composable
private fun FocusIndicator(
    x: Float,
    y: Float,
    isFocusing: Boolean,
    focusSuccess: Boolean?,
    isTracking: Boolean,
    isLocked: Boolean = false,
    isFading: Boolean = false,
    fadeStartTime: Long = 0L,
    modifier: Modifier = Modifier
) {
    // Position tracking with smooth animation
    val animatedX by animateFloatAsState(
        targetValue = x,
        animationSpec = if (isTracking) {
            spring(dampingRatio = Spring.DampingRatioNoBouncy, stiffness = Spring.StiffnessHigh)
        } else {
            tween(0)
        },
        label = "focus_x"
    )

    val animatedY by animateFloatAsState(
        targetValue = y,
        animationSpec = if (isTracking) {
            spring(dampingRatio = Spring.DampingRatioNoBouncy, stiffness = Spring.StiffnessHigh)
        } else {
            tween(0)
        },
        label = "focus_y"
    )

    // Mechanical clamp animation
    val scale = remember { Animatable(1.0f) }

    // v2.11.1 fix: Split into two separate LaunchedEffects
    // Problem: isFocusing in the key caused re-trigger during the 150ms shrink phase.
    // Fix: LaunchedEffect(focusSuccess) = clamp ONLY (not cancellable by isFocusing/isTracking)
    //      LaunchedEffect(isFocusing, isLocked) = search/locked states (guards against focusSuccess)

    // Clamp: keyed on focusSuccess only — runs to completion regardless of isFocusing changes
    LaunchedEffect(focusSuccess) {
        android.util.Log.d("FocusClamp", "LaunchedEffect(focusSuccess=$focusSuccess) triggered")
        when {
            focusSuccess == true -> {
                // Phase 1: Shrink to 0.75 (150ms)
                android.util.Log.d("FocusClamp", "Clamp START: shrink to 0.75")
                scale.animateTo(0.75f, tween(150, easing = FastOutLinearInEasing))
                // Phase 2: Bounce UP past 1.0 to 1.2 (120ms) — clearly visible expansion
                android.util.Log.d("FocusClamp", "Clamp MID: expanding to 1.2")
                scale.animateTo(1.2f, tween(120, easing = FastOutSlowInEasing))
                // Phase 3: Settle back to 1.0 (80ms)
                android.util.Log.d("FocusClamp", "Clamp SETTLE: returning to 1.0")
                scale.animateTo(1.0f, tween(80, easing = LinearOutSlowInEasing))
                android.util.Log.d("FocusClamp", "Clamp DONE: scale=${scale.value}")
            }
            else -> {
                android.util.Log.d("FocusClamp", "Clamp RESET (focusSuccess=$focusSuccess)")
                scale.snapTo(1.0f)
            }
        }
    }

    // Searching / Locked states — only when not in a focus-success clamp
    LaunchedEffect(isFocusing, isLocked) {
        if (focusSuccess != true) {
            when {
                isLocked -> scale.snapTo(1.0f)
                isFocusing -> scale.animateTo(1.2f, tween(300))
                else -> scale.snapTo(1.0f)
            }
        }
    }

    // Stroke width (Dwell: 2.5dp → 1.5dp)
    val animatedStrokeWidth by animateFloatAsState(
        targetValue = if (isFading) 1.5f else 2.5f,
        animationSpec = tween(500),
        label = "focus_stroke"
    )

    // Alpha — isFading takes priority over isTracking
    val alpha by animateFloatAsState(
        targetValue = when {
            isLocked -> 1.0f           // Lock state: always full alpha
            isFading -> 0.4f           // Dwell fade: applies even during tracking
            isTracking -> 1.0f         // Tracking (not fading): full alpha
            focusSuccess == false -> 0.6f
            else -> 1.0f
        },
        animationSpec = tween(500),
        label = "focus_alpha"
    )

    // Color: Cyan neon glow or Cyber Yellow (locked)
    val color = when {
        isLocked -> Color(0xFFFFD700)  // Cyber Yellow (#FFD700)
        else -> Color(0xFF00E5FF)      // Cyan neon glow (#00E5FF)
    }

    // Corner brackets appear only on success/locked
    val showCornerBrackets = focusSuccess == true || isLocked

    Canvas(modifier = modifier) {
        val size = 48.dp.toPx() * scale.value
        val halfSize = size / 2
        val strokeWidth = animatedStrokeWidth.dp.toPx()

        // Glow Layer (Bottom) - Neon effect
        drawRect(
            color = color,
            topLeft = Offset(animatedX - halfSize, animatedY - halfSize),
            size = Size(size, size),
            alpha = alpha * 0.3f,
            style = Stroke(width = strokeWidth * 2.4f)
        )

        // Core Layer (Top)
        drawRect(
            color = color,
            topLeft = Offset(animatedX - halfSize, animatedY - halfSize),
            size = Size(size, size),
            alpha = alpha,
            style = Stroke(width = strokeWidth)
        )

        // Corner Brackets (Conditional)
        if (showCornerBrackets) {
            val bracketLength = 12.dp.toPx()
            // Glow brackets
            drawCornerBrackets(
                centerX = animatedX,
                centerY = animatedY,
                halfSize = halfSize,
                bracketLength = bracketLength,
                strokeWidth = strokeWidth * 2.4f,
                color = color,
                alpha = alpha * 0.3f
            )
            // Core brackets
            drawCornerBrackets(
                centerX = animatedX,
                centerY = animatedY,
                halfSize = halfSize,
                bracketLength = bracketLength,
                strokeWidth = strokeWidth,
                color = color,
                alpha = alpha
            )
        }
    }
}
```

```kotlin
/**
 * Lock AE/AF at specified point (long press)
 * v2.9.9 Day 1-2: Traycer v2 AE/AF Lock
 */
private fun lockAEAF(x: Float, y: Float, viewWidth: Int, viewHeight: Int) {
    viewModelScope.launch {
        try {
            Log.i(TAG, "Locking AE/AF at ($x, $y)")

            // Clear tracking state if active (lock mode is exclusive)
            val wasTracking = _state.value.trackedObjectId != null
            if (wasTracking) {
                _state.update {
                    it.copy(
                        trackedObjectId = null,
                        trackedObjectType = null
                    )
                }
                Log.i(TAG, "Tracking cleared for AE/AF lock")
            }

            // Update state to show lock indicator (yellow square)
            _state.update {
                it.copy(
                    focusPoint = Pair(x, y),
                    isFocusing = true,
                    focusSuccess = null,
                    lockPoint = Pair(x, y),
                    isAEAFLocked = false  // Will be set to true after successful lock
                )
            }

            // Execute CameraX lock (persistent focus metering without auto-cancel)
            val result = cameraRepository.lockAEAF(x, y, viewWidth, viewHeight)

            if (result.isSuccess) {
                _state.update {
                    it.copy(
                        isFocusing = false,
                        focusSuccess = true,   // ← 應觸發 3-phase 動畫
                        isAEAFLocked = true
                    )
                }
                Log.i(TAG, "AE/AF locked successfully at ($x, $y)")
            } else {
                _state.update {
                    it.copy(
                        isFocusing = false,
                        focusSuccess = false,
                        lockPoint = null,
                        isAEAFLocked = false
                    )
                }
                Log.w(TAG, "AE/AF lock failed: ${result.exceptionOrNull()?.message}")
                delay(500)
                _state.update { it.copy(focusPoint = null, focusSuccess = null) }
            }

        } catch (e: Exception) {
            Log.e(TAG, "AE/AF lock error", e)
            _state.update {
                it.copy(
                    isFocusing = false,
                    focusSuccess = false,
                    lockPoint = null,
                    isAEAFLocked = false
                )
            }
        }
    }
}
```

#### Bug A：長按 B 點但鎖定在 A 點（座標錯誤）

1. 搜尋 ViewModel 中所有更新 `focusPoint` 的位置（ML Kit / Optical Flow 是否在 lockAEAF 期間寫入）

3. Logcat 加入 `lockAEAF` 座標確認傳入值是否正確

#### Bug B：AE/AF Lock 成功後無 3-phase 機械咬合動畫
**假設根因**：
- `focusSuccess = true` 和 `isAEAFLocked = true` 同在一個 `_state.update {}` 設置
- Compose Recomposition：`LaunchedEffect(focusSuccess)` 應觸發 → 但 `LaunchedEffect(isFocusing, isLocked)` 也同時觸發（isFocusing: true→false，isLocked 可能也變化）
- 兩個 LaunchedEffect 同時重啟，可能 `scale.snapTo(1.0f)` 覆蓋了動畫 start

**調查起點**：
1. 確認 FocusIndicator 的 `isLocked` 參數是映射 `isAEAFLocked` 還是其他 state
2. 若 `isLocked` 在 `LaunchedEffect(isFocusing, isLocked)` 的 key 中，且 `focusSuccess != true` guard 有問題 → 可能 `scale.snapTo(1.0f)` 取消了動畫

## ⚠️ 重要提醒（防止回歸）

| 規則 | 說明 |
|------|------|
| Mechanical clamp 必用 3-phase tween | Spring 動畫振幅不足（~5-13%），用戶看不到回彈 |
