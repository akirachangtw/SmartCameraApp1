**1. HistogramCalculator.kt**
```kotlin
class HistogramCalculator {
    fun calculateHistogram(image: ImageProxy): Histogram {
        // YUV_420_888 → Luminance channel
        val yPlane = image.planes[0]
        val histogram = IntArray(256) { 0 }

        // Sample every 4th pixel for performance
        for (y in 0 until image.height step 4) {
            for (x in 0 until image.width step 4) {
                val luminance = yPlane.buffer[y * rowStride + x].toInt() and 0xFF
                histogram[luminance]++
            }
        }

        return Histogram(
            data = histogram,
            totalPixels = (image.width / 4) * (image.height / 4)
        )
    }
}
```

**2. ExposureDetector.kt**
```kotlin
class ExposureDetector {
    fun analyze(histogram: Histogram, subjectBox: Rect?): ExposureAnalysis {
        // 高光比例 (240-255)
        val highlightRatio = histogram.data.sliceArray(240..255).sum() /
            histogram.totalPixels.toFloat()

        // 陰影比例 (0-30)
        val shadowRatio = histogram.data.sliceArray(0..30).sum() /
            histogram.totalPixels.toFloat()

        // 平均亮度
        val avgLuminance = histogram.data.indices.sumOf {
            it * histogram.data[it]
        } / histogram.totalPixels.toFloat()

        return ExposureAnalysis(
            isBacklit = detectBacklight(subjectBox, histogram),
            isOverexposed = highlightRatio > 0.15f,
            isLowLight = avgLuminance < 51, // 20% of 255
            highlightRatio = highlightRatio,
            shadowRatio = shadowRatio,
            avgLuminance = avgLuminance
        )
    }

    private fun detectBacklight(subjectBox: Rect?, histogram: Histogram): Boolean {
        // 需要主體區域與背景區域分別計算
        // 簡化版: 主體暗 + 整體亮度高 = 逆光
        if (subjectBox == null) return false

        val subjectAvgLuminance = calculateRegionLuminance(subjectBox)
        val backgroundAvgLuminance = histogram.avgLuminance

        return subjectAvgLuminance < 76 && // 主體 < 30%
               backgroundAvgLuminance > 178 // 背景 > 70%
    }
}
```

**1. LeadingLinesDetector.kt**
```kotlin
class LeadingLinesDetector(private val openCVManager: OpenCVManager) {

    fun detectLines(image: ImageProxy): List<Line> {
        val mat = ImageConverter.toMat(image)
        val gray = Mat()
        val edges = Mat()
        val lines = Mat()

        // 1. 灰階轉換
        Imgproc.cvtColor(mat, gray, Imgproc.COLOR_RGB2GRAY)

        // 2. Canny 邊緣偵測
        Imgproc.Canny(gray, edges, 50.0, 150.0)

        // 3. Hough Line Transform
        Imgproc.HoughLinesP(
            edges, lines,
            rho = 1.0,
            theta = Math.PI / 180,
            threshold = 50,
            minLineLength = 100.0,
            maxLineGap = 10.0
        )

        // 4. 篩選與分組
        val detectedLines = mutableListOf<Line>()
        for (i in 0 until lines.rows()) {
            val line = lines.get(i, 0)
            detectedLines.add(
                Line(
                    x1 = line[0].toFloat(),
                    y1 = line[1].toFloat(),
                    x2 = line[2].toFloat(),
                    y2 = line[3].toFloat()
                )
            )
        }

        // 5. 選擇主導線 (最長且接近對角線)
        return selectDominantLines(detectedLines)
    }

    private fun selectDominantLines(lines: List<Line>): List<Line> {
        return lines
            .sortedByDescending { it.length }
            .filter { line ->
                val angle = line.angle
                // 篩選對角線 (30-60°) 或水平線 (±10°)
                (angle in 30.0..60.0) || (angle in -10.0..10.0)
            }
            .take(3) // 最多 3 條主導線
    }
}
```

```kotlin
class SymmetryDetector {
    fun analyzeSymmetry(image: ImageProxy): SymmetryAnalysis {
        val mat = ImageConverter.toMat(image)
        val width = mat.width()
        val centerX = width / 2

        // 分割左右兩側
        val leftRegion = mat.submat(0, mat.height(), 0, centerX)
        val rightRegion = mat.submat(0, mat.height(), centerX, width)

        // 水平翻轉右側
        val rightFlipped = Mat()
        Core.flip(rightRegion, rightFlipped, 1)

        // 計算相似度 (SSIM or MSE)
        val similarity = calculateSimilarity(leftRegion, rightFlipped)

        // 亮度比較
        val leftAvgLuminance = Core.mean(leftRegion).`val`[0]
        val rightAvgLuminance = Core.mean(rightRegion).`val`[0]
        val luminanceDiff = abs(leftAvgLuminance - rightAvgLuminance)

        return SymmetryAnalysis(
            isSymmetric = similarity > 0.75 && luminanceDiff < 20,
            symmetryScore = similarity,
            luminanceDiff = luminanceDiff,
            suggestedShift = calculateShift(similarity)
        )
    }
}
```
