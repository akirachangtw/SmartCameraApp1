**位置**：`AnalyzeCompositionUseCase.kt` (Lines 495-569)

```kotlin
/**
 * 眼睛位置分析結果
 * v2.3.0: 專業人像構圖 - 眼睛應位於上三分線
 */
private data class EyePositionAnalysis(
    val hasEyeLandmarks: Boolean,        // 是否成功取得眼睛座標
    val averageEyeY: Float,              // 左右眼平均 Y 座標
    val upperThirdLine: Float,           // 上三分線位置
    val deviationFromIdeal: Float,       // 與理想位置的偏差
    val needsAdjustment: Boolean,        // 是否需要調整
    val isAboveIdeal: Boolean            // 是否高於理想位置
) {
    companion object {
        private const val EYE_POSITION_TOLERANCE_RATIO = 0.08f  // 容差 8%

        fun create(
            face: com.google.mlkit.vision.face.Face,
            imageHeight: Float
        ): EyePositionAnalysis {
            val leftEye = face.getLandmark(com.google.mlkit.vision.face.FaceLandmark.LEFT_EYE)
            val rightEye = face.getLandmark(com.google.mlkit.vision.face.FaceLandmark.RIGHT_EYE)

            // 檢查是否有眼睛座標
            if (leftEye == null || rightEye == null) {
                return EyePositionAnalysis(
                    hasEyeLandmarks = false,
                    averageEyeY = 0f,
                    upperThirdLine = imageHeight / 3f,
                    deviationFromIdeal = 0f,
                    needsAdjustment = false,
                    isAboveIdeal = false
                )
            }

            // 計算左右眼平均 Y 座標
            val avgEyeY = (leftEye.position.y + rightEye.position.y) / 2f

            // 上三分線（理想眼睛位置）
            val upperThird = imageHeight / 3f

            // 容差（8% 畫面高度）
            val tolerance = imageHeight * EYE_POSITION_TOLERANCE_RATIO

            // 計算偏差
            val deviation = avgEyeY - upperThird

            return EyePositionAnalysis(
                hasEyeLandmarks = true,
                averageEyeY = avgEyeY,
                upperThirdLine = upperThird,
                deviationFromIdeal = deviation,
                needsAdjustment = kotlin.math.abs(deviation) > tolerance,
                isAboveIdeal = avgEyeY < upperThird
            )
        }
    }
}
```

/**
 * 啟用 HDR 模式（透過曝光補償）
 *
 * v2.3.0: 逆光場景自動 HDR
 *
 * 注意：CameraX 沒有直接 HDR 控制，使用曝光補償作為替代
 */
override fun enableHDRMode() {
    try {
        val cameraControl = _cameraController.cameraControl
        if (cameraControl != null) {
            // 設定曝光補償 +1 stop
            cameraControl.setExposureCompensationIndex(1)
            Log.i(TAG, "HDR mode enabled (exposure compensation +1)")
        } else {
            Log.w(TAG, "Cannot enable HDR: CameraControl not available")
        }
    } catch (e: Exception) {
        Log.e(TAG, "Failed to enable HDR mode", e)
    }
}

/**
 * 停用 HDR 模式（重設曝光補償）
 *
 * v2.3.0: 逆光場景自動 HDR
 */
override fun disableHDRMode() {
    try {
        val cameraControl = _cameraController.cameraControl
        if (cameraControl != null) {
            // 重設曝光補償為 0
            cameraControl.setExposureCompensationIndex(0)
            Log.i(TAG, "HDR mode disabled (exposure compensation 0)")
        } else {
            Log.w(TAG, "Cannot disable HDR: CameraControl not available")
        }
    } catch (e: Exception) {
        Log.e(TAG, "Failed to disable HDR mode", e)
    }
}

/**
 * 檢查是否應啟用自動 HDR
 */
override fun shouldEnableAutoHDR(lightingState: LightingState): Boolean {
    return when (lightingState) {
        is LightingState.HighContrast -> lightingState.analysis.shouldUseHDR
        else -> false
    }
}
```

#### D. Auto HDR 觀察者邏輯
**位置**：`CameraViewModel.kt` (Lines 118-143)

/**
 * 處理自動 HDR 模式
 * v2.3.0: 逆光場景自動 HDR
 */
private fun handleAutoHDR(
    repository: local.mylab.smartcameraapp1.data.repository.CameraRepositoryImpl,
    lightingState: LightingState
) {
    viewModelScope.launch {
        overlayPreferences.isAutoHDREnabled.first().let { autoHDREnabled ->
            if (autoHDREnabled) {
                val shouldEnable = repository.shouldEnableAutoHDR(lightingState)

                if (shouldEnable && !_isAutoHDRActive.value) {
                    repository.enableHDRMode()
                    _isAutoHDRActive.value = true
                    Log.i(TAG, "Auto HDR activated")
                } else if (!shouldEnable && _isAutoHDRActive.value) {
                    repository.disableHDRMode()
                    _isAutoHDRActive.value = false
                    Log.i(TAG, "Auto HDR deactivated")
                }
            } else if (_isAutoHDRActive.value) {
                repository.disableHDRMode()
                _isAutoHDRActive.value = false
                Log.i(TAG, "Auto HDR disabled via settings")
            }
        }
    }
}
```

## 6. 重要技術備註

### 6.1 ML Kit Landmarks 限制
- **最佳效果**：正臉（±45° 旋轉範圍）
- **偵測失敗時**：自動降級為邊界框分析（無 landmark 數據）
- **效能開銷**：+10-20ms per frame（總分析時間 ~30-35ms）

### 6.2 CameraX 曝光補償
- **非真實 HDR**：使用 `setExposureCompensationIndex()` 模擬 HDR 效果
- **裝置相依**：實際效果取決於裝置相機硬體
- **補償範圍**：0 = 正常，+1 = 增加 1 stop 曝光

### 6.3 構圖評分系統
- **良好構圖閾值**：`overallScore >= 0.7` (70%)
- **權重總和**：必須為 1.0 (100%)
- **單一規則閾值**：0.7 為觸發提示的門檻

### 6.4 已知問題 & 改進方向

**未來改進**：
