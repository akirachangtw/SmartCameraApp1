// Lines 495-569 - EyePositionAnalysis Data Class
private data class EyePositionAnalysis(
    val hasEyeLandmarks: Boolean,
    val averageEyeY: Float,
    val upperThirdLine: Float,
    val deviationFromIdeal: Float,
    val needsAdjustment: Boolean,
    val isAboveIdeal: Boolean
) {
    companion object {
        private const val EYE_POSITION_TOLERANCE_RATIO = 0.08f

        fun create(face: com.google.mlkit.vision.face.Face, imageHeight: Float): EyePositionAnalysis {
            val leftEye = face.getLandmark(com.google.mlkit.vision.face.FaceLandmark.LEFT_EYE)
            val rightEye = face.getLandmark(com.google.mlkit.vision.face.FaceLandmark.RIGHT_EYE)
            
            if (leftEye == null || rightEye == null) {
                return EyePositionAnalysis(false, 0f, imageHeight / 3f, 0f, false, false)
            }
            
            val avgEyeY = (leftEye.position.y + rightEye.position.y) / 2f
            val upperThird = imageHeight / 3f
            val tolerance = imageHeight * EYE_POSITION_TOLERANCE_RATIO
            val deviation = avgEyeY - upperThird
            
            return EyePositionAnalysis(
                hasEyeLandmarks = true,
                averageEyeY = avgEyeY,
                upperThirdLine = upperThird,
                deviationFromIdeal = deviation,
                needsAdjustment = kotlin.math.abs(deviation) > tolerance,
                isAboveIdeal = avgEyeY < upperThird
            )
        }
    }
}

private fun handleAutoHDR(
    repository: local.mylab.smartcameraapp1.data.repository.CameraRepositoryImpl,
    lightingState: LightingState
) {
    viewModelScope.launch {
        overlayPreferences.isAutoHDREnabled.first().let { autoHDREnabled ->
            if (autoHDREnabled) {
                val shouldEnable = repository.shouldEnableAutoHDR(lightingState)
                
                if (shouldEnable && !_isAutoHDRActive.value) {
                    repository.enableHDRMode()
                    _isAutoHDRActive.value = true
                    Log.i(TAG, "Auto HDR activated")
                } else if (!shouldEnable && _isAutoHDRActive.value) {
                    repository.disableHDRMode()
                    _isAutoHDRActive.value = false
                    Log.i(TAG, "Auto HDR deactivated")
                }
            } else if (_isAutoHDRActive.value) {
                repository.disableHDRMode()
                _isAutoHDRActive.value = false
                Log.i(TAG, "Auto HDR disabled via settings")
            }
        }
    }
}
