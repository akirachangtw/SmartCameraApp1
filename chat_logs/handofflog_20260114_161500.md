## 1. 專案背景 (Context)

### 專案概述
SmartCameraApp 是一款 Android AI 攝影助手應用,使用 CameraX + ML Kit 提供即時構圖建議。

### 當前問題與解決方案
**問題**: v2.9.0-beta5 使用品牌判斷邏輯選擇解析度,但 OnePlus CPH2381 案例證明此方法有 14% 錯誤率(品牌相關但硬體不同)。

**解決方案**: v2.9.5 實作動態硬體檢測機制:
- 移除所有品牌/型號字串判斷
- 使用 Camera2 API 查詢實際硬體支援解析度
- 首次安裝時快取檢測結果至 SharedPreferences
- 提供診斷頁面 UI 供測試驗證

### 關鍵技術決策
1. **純硬體檢測**: 不依賴 `Build.MANUFACTURER` 或 `Build.MODEL`
2. **首次安裝快取**: 檢測僅執行一次,後續從快取讀取 (<5ms)
3. **版本化快取**: APP 升級時自動失效重新檢測
4. **Fail-safe 設計**: 任何失敗情況都 fallback 至 640x480

```kotlin
package com.akirachang.ai.photographer.data.source.camera

import android.content.Context
import android.content.SharedPreferences
import android.hardware.camera2.CameraCharacteristics
import android.hardware.camera2.CameraManager
import android.util.Log
import android.util.Size
import com.akirachang.ai.photographer.BuildConfig
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.util.concurrent.atomic.AtomicBoolean

/**
 * Camera Capability Detector
 * v2.9.5: Dynamic hardware-based resolution detection with caching
 *
 * Purpose:
 * - Detect camera hardware supported resolutions using Camera2 API
 * - Cache detection results to SharedPreferences for fast subsequent launches
 * - Remove all brand-based logic (proven 14% error rate with OnePlus case)
 *
 * Strategy:
 * - If device supports 800x600 → use 800x600 (Graphics ~84MB, better AI accuracy)
 * - If not → use largest resolution ≤800x600 (typically 640x480, Graphics ~46MB)
 *
 * Cache Invalidation:
 * - APP version upgrade (BuildConfig.VERSION_NAME changes)
 * - Cache schema version upgrade (CACHE_VERSION changes)
 * - Manual cache clear via clearCache()
 *
 * Thread Safety:
 * - Uses AtomicBoolean to prevent concurrent detection
 * - All detection runs on Dispatchers.IO
 *
 * @param context Application context for Camera2 API and SharedPreferences access
 */
class CameraCapabilityDetector(private val context: Context) {

    companion object {
        private const val TAG = "CameraCapabilityDetector"
        private const val PREFS_NAME = "camera_capability_cache"
        private const val KEY_CACHE_VERSION = "cache_version"
        private const val KEY_APP_VERSION = "app_version"
        private const val KEY_SUPPORTS_800x600 = "supports_800x600"
        private const val KEY_TARGET_WIDTH = "target_width"
        private const val KEY_TARGET_HEIGHT = "target_height"
        private const val KEY_FILTER_MAX_WIDTH = "filter_max_width"
        private const val KEY_FILTER_MAX_HEIGHT = "filter_max_height"

        private const val CACHE_VERSION = 1
        private const val TARGET_800x600_WIDTH = 800
        private const val TARGET_800x600_HEIGHT = 600
    }

    private val prefs: SharedPreferences = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
    private val isDetecting = AtomicBoolean(false)

    /**
     * Cached resolution strategy data class
     *
     * @property supports800x600 Whether device camera supports 800x600 resolution
     * @property targetResolution Target resolution to use for CameraX ResolutionSelector
     * @property filterMaxWidth Maximum width for resolution filter
     * @property filterMaxHeight Maximum height for resolution filter
     */
    data class CachedResolutionStrategy(
        val supports800x600: Boolean,
        val targetResolution: Size,
        val filterMaxWidth: Int,
        val filterMaxHeight: Int
    )

    /**
     * Detect camera capabilities and cache results
     * v2.9.5: Hardware-based detection without brand checks
     *
     * Process:
     * 1. Check if already detecting (in-flight guard)
     * 2. Query Camera2 API for supported YUV_420_888 resolutions
     * 3. Determine if 800x600 is supported
     * 4. Cache results to SharedPreferences
     * 5. Return strategy
     *
     * @return Result containing CachedResolutionStrategy or error
     */
    suspend fun detectAndCacheCapabilities(): Result<CachedResolutionStrategy> = withContext(Dispatchers.IO) {
        try {
            // In-flight guard: prevent concurrent detection
            if (!isDetecting.compareAndSet(false, true)) {
                Log.w(TAG, "[detect_already_running] Detection already in progress, skipping")
                return@withContext Result.failure(IllegalStateException("Detection already in progress"))
            }

            Log.i(TAG, "[detect_start] Starting camera capability detection")

            val cameraManager = context.getSystemService(Context.CAMERA_SERVICE) as CameraManager
            val cameraIdList = cameraManager.cameraIdList

            if (cameraIdList.isEmpty()) {
                Log.e(TAG, "[detect_fail] No cameras found on device")
                isDetecting.set(false)
                return@withContext Result.failure(IllegalStateException("No cameras found"))
            }

            // Use back camera (usually camera ID 0)
            val backCameraId = cameraIdList.firstOrNull { id ->
                val characteristics = cameraManager.getCameraCharacteristics(id)
                characteristics.get(CameraCharacteristics.LENS_FACING) == CameraCharacteristics.LENS_FACING_BACK
            } ?: cameraIdList[0]

            val characteristics = cameraManager.getCameraCharacteristics(backCameraId)
            val configMap = characteristics.get(CameraCharacteristics.SCALER_STREAM_CONFIGURATION_MAP)

            if (configMap == null) {
                Log.e(TAG, "[detect_fail] StreamConfigurationMap is null")
                isDetecting.set(false)
                return@withContext Result.failure(IllegalStateException("StreamConfigurationMap is null"))
            }

            // Query YUV_420_888 format supported resolutions
            val yuvFormat = android.graphics.ImageFormat.YUV_420_888
            val supportedSizes = configMap.getOutputSizes(yuvFormat)?.toList() ?: emptyList()

            Log.i(TAG, "[detect_info] Camera $backCameraId supports ${supportedSizes.size} YUV resolutions")

            // Check if 800x600 is supported
            val supports800x600 = supportedSizes.any { size ->
                size.width == TARGET_800x600_WIDTH && size.height == TARGET_800x600_HEIGHT
            }

            // Determine strategy based on hardware capability
            val strategy = if (supports800x600) {
                Log.i(TAG, "[detect_success] Device supports 800x600 → using 800x600 strategy")
                CachedResolutionStrategy(
                    supports800x600 = true,
                    targetResolution = Size(TARGET_800x600_WIDTH, TARGET_800x600_HEIGHT),
                    filterMaxWidth = 1280,
                    filterMaxHeight = 720
                )
            } else {
                Log.i(TAG, "[detect_success] Device does NOT support 800x600 → using 640x480 strategy")
                // Find largest resolution ≤800x600
                val acceptableResolutions = supportedSizes.filter { size ->
                    size.width <= TARGET_800x600_WIDTH && size.height <= TARGET_800x600_HEIGHT &&
                            size.width >= 480 && size.height >= 360
                }.sortedByDescending { it.width * it.height }

                val optimalResolution = acceptableResolutions.firstOrNull() ?: Size(640, 480)

                CachedResolutionStrategy(
                    supports800x600 = false,
                    targetResolution = optimalResolution,
                    filterMaxWidth = TARGET_800x600_WIDTH,
                    filterMaxHeight = TARGET_800x600_HEIGHT
                )
            }

            // Cache results
            prefs.edit().apply {
                putInt(KEY_CACHE_VERSION, CACHE_VERSION)
                putString(KEY_APP_VERSION, BuildConfig.VERSION_NAME)
                putBoolean(KEY_SUPPORTS_800x600, strategy.supports800x600)
                putInt(KEY_TARGET_WIDTH, strategy.targetResolution.width)
                putInt(KEY_TARGET_HEIGHT, strategy.targetResolution.height)
                putInt(KEY_FILTER_MAX_WIDTH, strategy.filterMaxWidth)
                putInt(KEY_FILTER_MAX_HEIGHT, strategy.filterMaxHeight)
                apply()
            }

            Log.i(TAG, "[cache_write] Cached strategy: supports800x600=${strategy.supports800x600}, " +
                    "target=${strategy.targetResolution.width}x${strategy.targetResolution.height}")

            isDetecting.set(false)
            Result.success(strategy)

        } catch (e: Exception) {
            Log.e(TAG, "[detect_fail] Exception during detection: ${e.message}", e)
            isDetecting.set(false)
            Result.failure(e)
        }
    }

    /**
     * Get cached resolution strategy
     * v2.9.5: Fast cache read (<5ms)
     *
     * Cache Validation:
     * - Check cache version matches CACHE_VERSION
     * - Check app version matches BuildConfig.VERSION_NAME
     * - Check all required keys exist
     *
     * @return CachedResolutionStrategy if valid cache exists, null otherwise
     */
    fun getCachedStrategy(): CachedResolutionStrategy? {
        val cacheVersion = prefs.getInt(KEY_CACHE_VERSION, -1)
        val appVersion = prefs.getString(KEY_APP_VERSION, null)

        // Cache validation
        if (cacheVersion != CACHE_VERSION) {
            Log.i(TAG, "[cache_miss] Cache version mismatch: cached=$cacheVersion, expected=$CACHE_VERSION")
            return null
        }

        if (appVersion != BuildConfig.VERSION_NAME) {
            Log.i(TAG, "[cache_miss] App version changed: cached=$appVersion, current=${BuildConfig.VERSION_NAME}")
            return null
        }

        // Check if all required keys exist
        if (!prefs.contains(KEY_SUPPORTS_800x600) || !prefs.contains(KEY_TARGET_WIDTH)) {
            Log.i(TAG, "[cache_miss] Incomplete cache data")
            return null
        }

        val strategy = CachedResolutionStrategy(
            supports800x600 = prefs.getBoolean(KEY_SUPPORTS_800x600, false),
            targetResolution = Size(
                prefs.getInt(KEY_TARGET_WIDTH, 640),
                prefs.getInt(KEY_TARGET_HEIGHT, 480)
            ),
            filterMaxWidth = prefs.getInt(KEY_FILTER_MAX_WIDTH, 800),
            filterMaxHeight = prefs.getInt(KEY_FILTER_MAX_HEIGHT, 600)
        )

        Log.i(TAG, "[cache_hit] Using cached strategy: supports800x600=${strategy.supports800x600}, " +
                "target=${strategy.targetResolution.width}x${strategy.targetResolution.height}")

        return strategy
    }

    /**
     * Check if capabilities are already cached
     *
     * @return true if valid cache exists, false otherwise
     */
    fun isCapabilitiesCached(): Boolean {
        return getCachedStrategy() != null
    }

    /**
     * Clear cache
     * v2.9.5: Force re-detection on next launch
     *
     * Use case:
     * - Testing/debugging
     * - User manually triggers re-detection
     */
    fun clearCache() {
        prefs.edit().clear().apply()
        Log.i(TAG, "[cache_clear] Cache cleared, will re-detect on next launch")
    }

    /**
     * Get diagnostic information for UI display
     * v2.9.5: For Settings → Camera Diagnostics section
     *
     * @return Map of diagnostic key-value pairs
     */
    fun getDiagnosticInfo(): Map<String, String> {
        val info = mutableMapOf<String, String>()

        // Device information
        info["Manufacturer"] = android.os.Build.MANUFACTURER
        info["Model"] = android.os.Build.MODEL
        info["SDK"] = android.os.Build.VERSION.SDK_INT.toString()

        // Cache information
        val cacheVersion = prefs.getInt(KEY_CACHE_VERSION, 0)
        val appVersion = prefs.getString(KEY_APP_VERSION, "N/A") ?: "N/A"
        val supports800x600 = prefs.getBoolean(KEY_SUPPORTS_800x600, false)
        val targetWidth = prefs.getInt(KEY_TARGET_WIDTH, 0)
        val targetHeight = prefs.getInt(KEY_TARGET_HEIGHT, 0)
        val filterMaxWidth = prefs.getInt(KEY_FILTER_MAX_WIDTH, 0)
        val filterMaxHeight = prefs.getInt(KEY_FILTER_MAX_HEIGHT, 0)

        info["Cache Version"] = if (cacheVersion > 0) cacheVersion.toString() else "N/A"
        info["App Version"] = appVersion
        info["Supports 800x600"] = if (cacheVersion > 0) supports800x600.toString() else "N/A"
        info["Target Resolution"] = if (targetWidth > 0) "${targetWidth}x${targetHeight}" else "N/A"
        info["Filter Max"] = if (filterMaxWidth > 0) "${filterMaxWidth}x${filterMaxHeight}" else "N/A"

        return info
    }
}
```

```kotlin
/**
 * Select device optimal resolution
 * v2.9.5: Cache-first with runtime fallback, NO brand checks
 */
private fun selectDeviceOptimalResolution(supportedSizes: List<Size>): Triple<Size, Int, Int> {
    // ✅ Step 1: Try to use cached strategy
    val cachedStrategy = capabilityDetector.getCachedStrategy()
    if (cachedStrategy != null) {
        Log.i(TAG, "[v2.9.5 Cache] Using cached strategy: " +
                "supports800x600=${cachedStrategy.supports800x600}, " +
                "target=${cachedStrategy.targetResolution.width}x${cachedStrategy.targetResolution.height}, " +
                "filter≤${cachedStrategy.filterMaxWidth}x${cachedStrategy.filterMaxHeight}")
        return Triple(
            cachedStrategy.targetResolution,
            cachedStrategy.filterMaxWidth,
            cachedStrategy.filterMaxHeight
        )
    }

    // ⚠️ Step 2: Cache miss - fall back to runtime detection
    Log.w(TAG, "[v2.9.5 Fallback] No cache, using runtime detection")

    val acceptableResolutions = supportedSizes.filter { s ->
        s.width <= 1280 && s.height <= 720 && s.width >= 480 && s.height >= 360
    }.sortedBy { it.width * it.height }

    val supports800x600 = acceptableResolutions.any { it.width == 800 && it.height == 600 }

    if (supports800x600) {
        Log.i(TAG, "[v2.9.5 Runtime] Found 800x600 in supported list → using 800x600")
        return Triple(Size(800, 600), 1280, 720)
    }

    val optimal = acceptableResolutions.lastOrNull { s -> s.width <= 800 && s.height <= 600 }
    if (optimal != null) {
        Log.i(TAG, "[v2.9.5 Runtime] Using optimal: ${optimal.width}x${optimal.height}")
        return Triple(optimal, 800, 600)
    }

    Log.w(TAG, "[v2.9.5 Fallback] Using ultimate fallback: 640x480")
    return Triple(Size(640, 480), 800, 600)
}
```

```kotlin
/**
 * Detect camera capabilities if needed
 * v2.9.5: Triggered after camera permission granted
 */
fun detectCameraCapabilitiesIfNeeded() {
    viewModelScope.launch(Dispatchers.IO) {
        try {
            if (capabilityDetector.isCapabilitiesCached()) {
                Log.i(TAG, "[v2.9.5] Capabilities already cached, skipping detection")
                return@launch
            }

            Log.i(TAG, "[v2.9.5] Detecting camera capabilities...")
            val result = capabilityDetector.detectAndCacheCapabilities()

            result.onSuccess { strategy ->
                Log.i(TAG, "[v2.9.5] Detection success: " +
                        "supports800x600=${strategy.supports800x600}, " +
                        "target=${strategy.targetResolution.width}x${strategy.targetResolution.height}")
            }.onFailure { e ->
                Log.e(TAG, "[v2.9.5] Detection failed: ${e.message}", e)
            }
        } catch (e: Exception) {
            Log.e(TAG, "[v2.9.5] Unexpected error during detection: ${e.message}", e)
        }
    }
}
```
