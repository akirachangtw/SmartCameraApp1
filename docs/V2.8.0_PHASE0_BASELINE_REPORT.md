### 9.2 資料分析腳本

**FPS 統計** (Python):
```python
# tools/analyze_phase0_fps.py
import re
import statistics

fps_values = []
with open('phase0_pixel9a_perf.log', 'r') as f:
    for line in f:
        match = re.search(r'FPS=([\d.]+)', line)
        if match:
            fps_values.append(float(match.group(1)))

print(f"Average FPS: {statistics.mean(fps_values):.1f}")
print(f"Min FPS: {min(fps_values):.1f}")
print(f"Max FPS: {max(fps_values):.1f}")
print(f"Std Dev: {statistics.stdev(fps_values):.1f}")
```

**Memory 統計** (Python):
```python
# tools/analyze_phase0_memory.py
import re
import statistics

native_mb = []
java_mb = []
total_mb = []

with open('phase0_pixel9a_perf.log', 'r') as f:
    for line in f:
        match = re.search(r'Native=(\d+)MB, Java=(\d+)MB, Total=(\d+)MB', line)
        if match:
            native_mb.append(int(match.group(1)))
            java_mb.append(int(match.group(2)))
            total_mb.append(int(match.group(3)))

print(f"Native Heap: avg={statistics.mean(native_mb):.0f}MB, max={max(native_mb)}MB")
print(f"Java Heap: avg={statistics.mean(java_mb):.0f}MB, max={max(java_mb)}MB")
print(f"Total PSS: avg={statistics.mean(total_mb):.0f}MB, max={max(total_mb)}MB")
print(f"Memory Growth: {total_mb[-1] - total_mb[0]}MB over {len(total_mb)} samples")
```
