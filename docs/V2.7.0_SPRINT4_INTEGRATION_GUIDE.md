#### 步驟 1: 修改 GuidelineAnalyzer 支援 Bitmap 輸入

```kotlin
// GuidelineAnalyzer.kt
class GuidelineAnalyzer(
    private val onAnalysisResult: (GuidelineAnalysisResult) -> Unit
) {
    /**
     * Analyze bitmap for guideline lines
     *
     * Converts Bitmap to Mat and performs Hough Line Transform
     */
    fun analyzeBitmap(bitmap: Bitmap): GuidelineAnalysisResult {
        val startTime = System.nanoTime()

        try {
            // Convert Bitmap to Mat (grayscale)
            val grayMat = bitmapToGrayMat(bitmap)

            // Detect guideline lines
            val result = detectGuidelineLines(
                grayMat = grayMat,
                frameWidth = bitmap.width,
                frameHeight = bitmap.height
            )

            // Release Mat
            grayMat.release()

            // Calculate processing time
            val elapsedMillis = (System.nanoTime() - startTime) / 1_000_000

            return result.copy(processingTimeMs = elapsedMillis)

        } catch (e: Exception) {
            Log.e(TAG, "Bitmap analysis failed", e)
            return GuidelineAnalysisResult.EMPTY
        }
    }

    private fun bitmapToGrayMat(bitmap: Bitmap): Mat {
        val mat = Mat()
        val bmp32 = bitmap.copy(Bitmap.Config.ARGB_8888, true)
        Utils.bitmapToMat(bmp32, mat)

        // Convert to grayscale
        val grayMat = Mat()
        Imgproc.cvtColor(mat, grayMat, Imgproc.COLOR_RGBA2GRAY)

        mat.release()
        bmp32.recycle()

        return grayMat
    }
}
```

```kotlin
import com.akirachang.ai.photographer.data.source.analysis.GuidelineAnalyzer
import kotlinx.coroutines.flow.sample

class CameraViewModel @Inject constructor(
    private val cameraRepository: CameraRepository,
    // ... other dependencies
) : ViewModel() {

    // Guideline analysis state (v2.7.0 Sprint 4)
    private val _guidelineState = MutableStateFlow<GuidelineAnalysisResult?>(null)
    val guidelineState: StateFlow<GuidelineAnalysisResult?> = _guidelineState.asStateFlow()

    // GuidelineAnalyzer instance
    private val guidelineAnalyzer = GuidelineAnalyzer { result ->
        _guidelineState.value = result
    }

    init {
        // ... existing initialization

        // v2.7.0 Sprint 4: OpenCV Guideline Analysis
        observePreviewForGuidelineAnalysis()
    }

    /**
     * Observe preview bitmap for OpenCV guideline analysis
     *
     * Uses existing previewBitmap StateFlow to trigger analysis at 10 FPS
     */
    @OptIn(FlowPreview::class)
    private fun observePreviewForGuidelineAnalysis() {
        previewBitmap
            .filterNotNull()
            .sample(100) // 10 FPS (same as preview stream)
            .onEach { bitmap ->
                try {
                    // Trigger OpenCV analysis on background thread
                    withContext(Dispatchers.Default) {
                        val result = guidelineAnalyzer.analyzeBitmap(bitmap)
                        _guidelineState.value = result
                    }
                } catch (e: Exception) {
                    Log.e(TAG, "Guideline analysis failed", e)
                }
            }
            .launchIn(viewModelScope)
    }
}
```

```kotlin
class CameraRepositoryImpl @Inject constructor(
    @ApplicationContext private val context: Context,
```

## 🚨 常見問題與解決

### 問題 1: Bitmap → Mat 轉換錯誤

**錯誤訊息**:
```
E/GuidelineAnalyzer: Failed to convert Bitmap to Mat
```

**解決**:
```kotlin
// 確保 Bitmap 格式正確
val bmp32 = bitmap.copy(Bitmap.Config.ARGB_8888, true)
Utils.bitmapToMat(bmp32, mat)
```

### 問題 2: 效能不達標 (> 30ms)

**解決方案**:
1. 降低解析度:
   ```kotlin
   val scaledBitmap = Bitmap.createScaledBitmap(
       bitmap,
       bitmap.width / 2,
       bitmap.height / 2,
       true
   )
   ```

2. 增加 Hough threshold:
   ```kotlin
   private const val HOUGH_THRESHOLD = 100  // 從 50 增加到 100
   ```

3. 減少最大線條數:
   ```kotlin
   private const val MAX_LINES = 5  // 從 10 減少到 5
   ```

### 問題 3: 檢測線條太多 (雜訊)

**解決**:
1. 增加最小線條強度:
   ```kotlin
   private const val MIN_LINE_STRENGTH = 0.5f  // 從 0.3 增加到 0.5
   ```

2. 增加最小線長:
   ```kotlin
   private const val HOUGH_MIN_LINE_LENGTH = 100.0  // 從 50 增加到 100
   ```

### 問題 4: 檢測不到線條

**解決**:
1. 降低 Canny threshold:
   ```kotlin
   private const val CANNY_THRESHOLD_1 = 30.0  // 從 50 降到 30
   private const val CANNY_THRESHOLD_2 = 100.0  // 從 150 降到 100
   ```

2. 降低 Hough threshold:
   ```kotlin
   private const val HOUGH_THRESHOLD = 30  // 從 50 降到 30
   ```

```kotlin
private fun getOptimalBitmapSize(bitmap: Bitmap): Pair<Int, Int> {
    val maxDimension = 1280  // 根據裝置性能調整

    return if (bitmap.width > maxDimension || bitmap.height > maxDimension) {
        val scale = maxDimension.toFloat() / max(bitmap.width, bitmap.height)
        Pair(
```
