## 專案概述

### 功能目標

為相機預覽新增即時構圖引導線(Guideline Overlay),透過 OpenCV 分析相機畫面,動態顯示:
- **九宮格線** (Rule of Thirds)
- **中心安全區** (Center Safe Area)
- **水平線偵測** (Horizon Detection,可選)

### 技術方案

- **CameraX ImageAnalysis** → OpenCV Mat 分析
- **StateFlow** 傳遞 GuidelineResult
- **Compose Canvas** 繪製 Overlay
- **FrameToViewTransform** 處理座標映射

### 關鍵約束

1. **效能要求**: FPS ≥ 24,analyzer 耗時 < 30ms (P95)
2. **記憶體限制**: Mat 重用池,避免 GC 抖動
3. **相容性**: 支援 Android API 24-36,模擬器 + 實機
4. **回滾能力**: Feature flag 三段式控制

### Task 1.3: FrameToViewTransform 座標映射

**優先級**: P0 (必須)

#### 實作步驟

1. **建立 FrameToViewTransform 資料類別**
   ```kotlin
   // domain/model/FrameToViewTransform.kt
   package com.akirachang.ai.photographer.domain.model

   import android.graphics.Matrix
   import androidx.compose.ui.geometry.Offset
   import androidx.compose.ui.geometry.Rect
   import androidx.compose.ui.layout.ContentScale

   /**
    * Frame to View 座標映射轉換器
    *
    * 將 ImageAnalysis buffer 座標(640x480) 映射到 Compose View 座標
    *
    * 參考: ML Kit MlKitAnalyzer 的 COORDINATE_SYSTEM_VIEW_REFERENCED 實作
    *
    * @param imageWidth Analysis buffer 寬度 (e.g., 640)
    * @param imageHeight Analysis buffer 高度 (e.g., 480)
    * @param viewWidth Compose View 寬度
    * @param viewHeight Compose View 高度
    * @param contentScale Compose ContentScale (Crop or Fit)
    * @param rotationDegrees 旋轉角度 (0, 90, 180, 270)
    * @param isFrontCamera 是否為前鏡頭(需 mirror)
    */
   data class FrameToViewTransform(
       val imageWidth: Int,
       val imageHeight: Int,
       val viewWidth: Int,
       val viewHeight: Int,
       val contentScale: ContentScale,
       val rotationDegrees: Int,
       val isFrontCamera: Boolean
   ) {
       private val transformMatrix: Matrix by lazy {
           calculateTransformMatrix()
       }

       /**
        * 映射 buffer 座標點到 view 座標點
        */
       fun mapPoint(bufferX: Float, bufferY: Float): Offset {
           val point = floatArrayOf(bufferX, bufferY)
           transformMatrix.mapPoints(point)
           return Offset(point[0], point[1])
       }

       /**
        * 映射 buffer 矩形到 view 矩形
        */
       fun mapRect(bufferRect: Rect): Rect {
           // 四個角點映射
           val topLeft = mapPoint(bufferRect.left, bufferRect.top)
           val topRight = mapPoint(bufferRect.right, bufferRect.top)
           val bottomLeft = mapPoint(bufferRect.left, bufferRect.bottom)
           val bottomRight = mapPoint(bufferRect.right, bufferRect.bottom)

           // 找出映射後的邊界
           val left = minOf(topLeft.x, topRight.x, bottomLeft.x, bottomRight.x)
           val top = minOf(topLeft.y, topRight.y, bottomLeft.y, bottomRight.y)
           val right = maxOf(topLeft.x, topRight.x, bottomLeft.x, bottomRight.x)
           val bottom = maxOf(topLeft.y, topRight.y, bottomLeft.y, bottomRight.y)

           return Rect(left, top, right, bottom)
       }

       /**
        * 計算轉換矩陣
        *
        * 步驟:
        * 1. Rotation: 根據 rotationDegrees 旋轉
        * 2. Scale: 根據 ContentScale (Crop/Fit) 縮放
        * 3. Translate: 置中對齊
        * 4. Mirror: 前鏡頭水平翻轉
        */
       private fun calculateTransformMatrix(): Matrix {
           val matrix = Matrix()

           // Step 1: 處理旋轉後的尺寸
           val (rotatedWidth, rotatedHeight) = when (rotationDegrees) {
               90, 270 -> imageHeight to imageWidth
               else -> imageWidth to imageHeight
           }

           // Step 2: 計算縮放比例
           val scaleX: Float
           val scaleY: Float
           val translateX: Float
           val translateY: Float

           when (contentScale) {
               ContentScale.Crop -> {
                   // FILL_CENTER 行為: 縮放到填滿,可能裁切
                   val scale = maxOf(
                       viewWidth.toFloat() / rotatedWidth,
                       viewHeight.toFloat() / rotatedHeight
                   )
                   scaleX = scale
                   scaleY = scale

                   // 置中對齊
                   translateX = (viewWidth - rotatedWidth * scale) / 2f
                   translateY = (viewHeight - rotatedHeight * scale) / 2f
               }
               ContentScale.Fit -> {
                   // FIT_CENTER 行為: 縮放到適應,可能有黑邊
                   val scale = minOf(
                       viewWidth.toFloat() / rotatedWidth,
                       viewHeight.toFloat() / rotatedHeight
                   )
                   scaleX = scale
                   scaleY = scale

                   translateX = (viewWidth - rotatedWidth * scale) / 2f
                   translateY = (viewHeight - rotatedHeight * scale) / 2f
               }
               else -> {
                   scaleX = viewWidth.toFloat() / rotatedWidth
                   scaleY = viewHeight.toFloat() / rotatedHeight
                   translateX = 0f
                   translateY = 0f
               }
           }

           // Step 3: 建立轉換矩陣
           // 順序很重要: Translate → Scale → Rotate → Mirror

           // 3.1 平移到旋轉中心
           matrix.postTranslate(-imageWidth / 2f, -imageHeight / 2f)

           // 3.2 旋轉
           matrix.postRotate(rotationDegrees.toFloat())

           // 3.3 前鏡頭水平翻轉
           if (isFrontCamera) {
               matrix.postScale(-1f, 1f)
           }

           // 3.4 縮放
           matrix.postScale(scaleX, scaleY)

           // 3.5 平移到目標位置
           matrix.postTranslate(
               viewWidth / 2f + translateX,
               viewHeight / 2f + translateY
           )

           return matrix
       }
   }
   ```

   ```kotlin
   @SuppressLint("UnsafeOptInUsageError")
   fun ImageProxy.toGrayMat(): Mat {
       val yPlane = planes[0]
       val yBuffer = yPlane.buffer

       // 建立 Mat
       val mat = Mat(height, width, CvType.CV_8UC1)

       // 複製 Y plane 資料到 Mat
       val yData = ByteArray(yBuffer.remaining())
       yBuffer.get(yData)
       mat.put(0, 0, yData)

       return mat
   ```

```kotlin
    private val _resultFlow = MutableStateFlow<GuidelineResult?>(null)
    val resultFlow: StateFlow<GuidelineResult?> = _resultFlow.asStateFlow()

    private var frameCount = 0L

    override fun analyze(image: ImageProxy) {
        try {
            frameCount++

            // 每 2 幀分析一次 (降頻)
            if (frameCount % 2 != 0L) {
                image.close()
                return
            }

            val mat = image.toGrayMat()

            // ✅ OpenCV 分析: 九宮格線條
            val width = mat.cols()
            val height = mat.rows()

            val ruleOfThirdsLines = listOf(
                // 垂直線 1/3
                Line(
                    start = Offset(width / 3f, 0f),
                    end = Offset(width / 3f, height.toFloat())
                ),
                // 垂直線 2/3
                Line(
                    start = Offset(width * 2f / 3f, 0f),
                    end = Offset(width * 2f / 3f, height.toFloat())
                ),
                // 水平線 1/3
                Line(
                    start = Offset(0f, height / 3f),
                    end = Offset(width.toFloat(), height / 3f)
                ),
                // 水平線 2/3
                Line(
                    start = Offset(0f, height * 2f / 3f),
                    end = Offset(width.toFloat(), height * 2f / 3f)
                )
            )

            // TODO Phase 2.1: 水平線偵測 (Hough Line Transform)
            // val horizonLine = detectHorizonLine(mat)

            val result = GuidelineResult(
```

```kotlin
// data/source/analysis/GuidelineAnalyzer.kt (優化版)
class GuidelineAnalyzer(
    private val analyzeEveryNFrames: Int = 2
) : ImageAnalysis.Analyzer {

    // ✅ Mat 重用池 (最多 3 個)
    private val matPool = ArrayDeque<Mat>(3)

    private fun acquireMat(width: Int, height: Int): Mat {
        synchronized(matPool) {
            return matPool.removeFirstOrNull()?.apply {
                if (cols() != width || rows() != height) {
                    release()
                    return Mat(height, width, CvType.CV_8UC1)
                }
            } ?: Mat(height, width, CvType.CV_8UC1)
        }
    }

    private fun releaseMat(mat: Mat) {
        synchronized(matPool) {
            if (matPool.size < 3) {
                matPool.addLast(mat)
            } else {
                mat.release()
            }
        }
    }

    override fun analyze(image: ImageProxy) {
        try {
            frameCount++

            if (frameCount % analyzeEveryNFrames != 0L) {
                image.close()
                return
            }

            // ✅ 從池中取得 Mat
            val mat = acquireMat(image.width, image.height)

            try {
                // 複製資料到重用的 Mat
                val yPlane = image.planes[0]
                val yData = ByteArray(yPlane.buffer.remaining())
                yPlane.buffer.get(yData)
                mat.put(0, 0, yData)

                // OpenCV 分析...
                val result = GuidelineResult(...)
                _resultFlow.value = result

            } finally {
                // ✅ 歸還到池中
                releaseMat(mat)
            }

        } catch (e: Exception) {
            Log.e(TAG, "Analysis failed", e)
        } finally {
            image.close()
        }
    }

    // ✅ Cleanup
    fun cleanup() {
        synchronized(matPool) {
            matPool.forEach { it.release() }
            matPool.clear()
        }
    }
}
```
