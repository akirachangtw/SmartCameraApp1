步驟 2c：在 CameraViewModel 中新增 runNewFocusPipeline() private 函式（在 autoFocusOnSalientTarget() 之後）：

kotlin

/**
 * v2.12.0-15: New focus pipeline (Step 5b).
 * Pipeline: DetectionResult → GeometryMapper → TargetTracker → EyeAfPolicyResolver
 *           → FocusController → FocusStateMachine → FocusUiState
 *
 * This function is called from autoFocusOnSalientTarget() when newFocusPipelineEnabled=true.
 * It does NOT replace focusOnPoint() — manual tap still uses the legacy path.
 */
private fun runNewFocusPipeline() {
    val currentState = _state.value
    val nowMs = System.currentTimeMillis()

    // ── Step A: Get current DetectionResult ──────────────────────────────────
    val detectionResult = currentState.detectionResult
    if (detectionResult == null) {
        // No detection — update tracker with empty list
        val trackedState = focusTargetTracker.update(emptyList(), nowMs)
        val emptyDecision = EyeAfDecision(
            anchor = MeteringAnchor.NONE,
            visualState = VisualTargetState.None,
            shouldTriggerFocus = false,
            shouldTriggerMetering = false,
            allowOpticalState = false,
        )
        val focusInput = FocusStateInput(
            trackedTargetState = trackedState,
            eyeAfDecision = emptyDecision,
            focusControllerResult = null,
            userIntent = buildUserIntentState(currentState),
            systemEvent = SystemEvent.NORMAL_UPDATE,
            nowMs = nowMs,
        )
        val newFocusUiState = focusStateMachine.reduce(focusInput)
        _state.update { it.copy(focusUiState = newFocusUiState) }
        return
    }

    // ── Step B: Build FocusTarget list from DetectionResult ──────────────────
    // Convert DetectionResult faces/objects to FocusTarget list
    val rawTargets = buildFocusTargetsFromDetection(detectionResult, currentState, nowMs)

    // ── Step C: TargetTracker.update() ───────────────────────────────────────
    val trackedState = focusTargetTracker.update(rawTargets, nowMs)

    // ── Step D: Build LensCapability ─────────────────────────────────────────
    val lensCapability = buildLensCapability(currentState)

    // ── Step E: Build UserIntentState ────────────────────────────────────────
    val userIntent = buildUserIntentState(currentState)

    // ── Step F: EyeAfPolicyResolver.resolve() ────────────────────────────────
    val policyInput = EyeAfPolicyInput(
        primaryTarget = trackedState.primaryTarget,
        lensCapability = lensCapability,
        userIntent = userIntent,
        nowMs = nowMs,
        lastFocusTriggerMs = lastNewPipelineTriggerMs,
    )
    val eyeAfDecision = eyeAfPolicyResolver.resolve(policyInput)

    // ── Step G: FocusController.apply() (async, launch coroutine) ────────────
    var controllerResult: FocusControllerResult? = null
    if (eyeAfDecision.shouldTriggerFocus || eyeAfDecision.shouldTriggerMetering) {
        val geometryContext = buildGeometryContext(currentState)
        viewModelScope.launch {
            try {
                val result = cameraXFocusController.apply(
                    decision = eyeAfDecision,
                    primaryTarget = trackedState.primaryTarget,
                    geometryContext = geometryContext,
                )
                lastNewPipelineTriggerMs = nowMs
                Log.d(TAG, "[v2.12.0-15] FocusController result: triggerSent=${result.triggerSent}, optical=${result.opticalResult}")
            } catch (e: Exception) {
                Log.w(TAG, "[v2.12.0-15] FocusController.apply() failed: ${e.message}")
            }
        }
    }

    // ── Step H: FocusStateMachine.reduce() ───────────────────────────────────
    val focusInput = FocusStateInput(
        trackedTargetState = trackedState,
        eyeAfDecision = eyeAfDecision,
        focusControllerResult = controllerResult,
        userIntent = userIntent,
        systemEvent = SystemEvent.NORMAL_UPDATE,
        nowMs = nowMs,
    )
    val newFocusUiState = focusStateMachine.reduce(focusInput)

    // ── Step I: Update state ──────────────────────────────────────────────────
    _state.update { it.copy(focusUiState = newFocusUiState) }
    Log.d(TAG, "[v2.12.0-15] Pipeline: visual=${newFocusUiState.visualTargetState}, lens=${newFocusUiState.lensFocusState}")
}

/**
 * v2.12.0-15: Build FocusTarget list from current DetectionResult.
 * Converts ML Kit face/object detections to domain FocusTarget model.
 */
private fun buildFocusTargetsFromDetection(
    detectionResult: com.akirachang.ai.photographer.domain.model.DetectionResult,
    currentState: CameraState,
    nowMs: Long,
): List<FocusTarget> {
    val targets = mutableListOf<FocusTarget>()

    // Convert faces
    detectionResult.faces.forEachIndexed { index, face ->
        val rectView = face.boundingBoxView ?: return@forEachIndexed
        val stableId = if (face.trackingId != null) "face_${face.trackingId}" else "face_gen_${nowMs}_$index"
        targets.add(
            FocusTarget(
                stableId = stableId,
                type = TargetType.FACE,
                rectImage = face.boundingBoxImage ?: rectView,
                rectView = rectView,
                leftEyeView = face.leftEyeView,
                rightEyeView = face.rightEyeView,
                confidence = face.confidence,
                priorityScore = face.confidence * 1.5f,  // faces have higher priority
                trackingId = face.trackingId,
                firstSeenMs = nowMs,
                lastSeenMs = nowMs,
                source = TargetSource.ML_KIT_FACE,
            )
        )
    }

    // Convert objects
    detectionResult.objects.forEachIndexed { index, obj ->
        val rectView = obj.boundingBoxView ?: return@forEachIndexed
        val stableId = if (obj.trackingId != null) "object_${obj.trackingId}" else "object_gen_${nowMs}_$index"
        targets.add(
            FocusTarget(
                stableId = stableId,
                type = TargetType.OBJECT,
                rectImage = obj.boundingBoxImage ?: rectView,
                rectView = rectView,
                leftEyeView = null,
                rightEyeView = null,
                confidence = obj.confidence,
                priorityScore = obj.confidence,
                trackingId = obj.trackingId,
                firstSeenMs = nowMs,
                lastSeenMs = nowMs,
                source = TargetSource.ML_KIT_OBJECT,
            )
        )
    }

    return targets
}

/**
 * v2.12.0-15: Build LensCapability from current camera state.
 */
private fun buildLensCapability(currentState: CameraState): LensCapability {
    return LensCapability(
        isFrontCamera = currentState.currentCamera == com.akirachang.ai.photographer.domain.model.CameraLens.FRONT,
        isFixedFocus = currentState.isFixedFocusLens,
        supportsAutoFocus = !currentState.isFixedFocusLens,
        supportsContinuousAf = !currentState.isFixedFocusLens,
        supportsAeAfLock = true,
    )
}

/**
 * v2.12.0-15: Build UserIntentState from current camera state.
 */
private fun buildUserIntentState(currentState: CameraState): UserIntentState {
    val tapPoint = currentState.focusPoint?.let {
        android.graphics.PointF(it.first, it.second)
    }
    val isTapActive = currentState.isAEAFLocked ||
        (currentState.focusPoint != null && currentState.afState is AfState.Locked)
    return UserIntentState(
        tapPointView = tapPoint,
        tapTimestampMs = if (tapPoint != null) System.currentTimeMillis() else null,
        isTapToFocusActive = isTapActive,
        isAeAfLockRequested = currentState.isAEAFLocked,
    )
}

/**
 * v2.12.0-15: Build GeometryContext from current camera state.
 */
private fun buildGeometryContext(currentState: CameraState): GeometryContext {
    return GeometryContext(

        imageHeight = currentState.imageHeight,
        previewWidth = previewWidth,
        previewHeight = previewHeight,

        isMirroredPreview = currentState.currentCamera == com.akirachang.ai.photographer.domain.model.CameraLens.FRONT,
    )
}

### 架構師說明：為什麼採用「並行雙軌 + Feature Flag」策略？

| 決策 | 理由 |
|------|------|
| `newFocusPipelineEnabled = false`（預設關閉） | 確保 build 不破壞現有功能，可安全 merge 到主分支 |
| 不移除舊邏輯 | 舊路徑是 fallback，新路徑驗證通過後再移除（Step 6） |
| `CameraScreen.kt` 只加 placeholder | 避免 UI 視覺層在 pipeline 未完全驗證前就改變行為 |
| `runNewFocusPipeline()` 獨立函式 | 清晰的邊界，方便 Code Review 和後續移除舊邏輯 |
| `FocusController.apply()` 在 coroutine 中非同步執行 | 避免阻塞 `autoFocusOnSalientTarget()` 的主執行緒呼叫 |
