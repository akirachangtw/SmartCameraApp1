package com.akirachang.ai.photographer.data.focus.policy

import android.util.Log
import com.akirachang.ai.photographer.domain.focus.policy.EyeAfPolicyResolver
import com.akirachang.ai.photographer.domain.model.focus.EyeAfDecision
import com.akirachang.ai.photographer.domain.model.focus.EyeAfPolicyInput
import com.akirachang.ai.photographer.domain.model.focus.EyeAfPolicyOutput

/**
 * Default Eye-AF policy resolver.
 * v2.12.0-15 / §17.10 決策表
 *
 * Decision table:
 * ┌─────────────────────────────┬──────────────────────────────┐
 * │ Condition                   │ Decision                     │
 * ├─────────────────────────────┼──────────────────────────────┤
 * │ isFrontCamera && manual tap │ Reject (front cam no tap)    │
 * │ isFixedFocus                │ SyntheticLock (bypass HAL)   │
 * │ inCooldown                  │ Throttle (skip this frame)   │
 * │ otherwise                   │ Allow (normal HAL path)      │
 * └─────────────────────────────┴──────────────────────────────┘
 */
class DefaultEyeAfPolicyResolver : EyeAfPolicyResolver {

    companion object {
        private const val TAG = "DefaultEyeAfPolicyResolver"
    }

    override fun resolve(input: EyeAfPolicyInput): EyeAfPolicyOutput {
        // Rule 1: Front camera prohibits manual tap focus
        if (input.isFrontCamera && input.isManualTap) {
            Log.i(TAG, "[v2.12.0-15] Reject: front camera manual tap not allowed")
            return EyeAfPolicyOutput(decision = EyeAfDecision.Reject, reason = "front_camera_no_manual_tap")
        }

        // Rule 2: Fixed-focus lens — bypass HAL, use synthetic visual lock
        if (input.lensCapability.isFixedFocus) {
            Log.i(TAG, "[v2.12.0-15] SyntheticLock: fixed-focus lens bypass HAL")
            return EyeAfPolicyOutput(decision = EyeAfDecision.SyntheticLock, reason = "fixed_focus_bypass")
        }

        // Rule 3: Cooldown throttle — skip if within post-lock cooldown window
        if (input.isInCooldown) {
            Log.d(TAG, "[v2.12.0-15] Throttle: within post-lock cooldown window")
            return EyeAfPolicyOutput(decision = EyeAfDecision.Throttle, reason = "post_lock_cooldown")
        }

        // Rule 4: Default — allow normal HAL focus path
        Log.d(TAG, "[v2.12.0-15] Allow: normal HAL focus path")
        return EyeAfPolicyOutput(decision = EyeAfDecision.Allow, reason = "normal_hal_path")
    }
}

### 架構師說明：`EyeAfDecision.Throttle` 的映射策略

現有 `FocusOrchestrator` 沒有 `Throttle` 這個 `FocusDecision`，因為 throttle 邏輯由 `eyeAfThrottle`（在 `CameraViewModel` 中）處理。因此在 Thin Adapter 中，`Throttle → Allow`，讓 `eyeAfThrottle` 繼續承擔 cooldown 的實際攔截。這樣可以保持 `CameraViewModel` 的呼叫端不受影響，符合「最小改動」原則。

若 `EyeAfPolicyModels.kt` 中的 `EyeAfDecision` 沒有 `Throttle` 這個值，請以該檔案的實際定義為準，不要自行新增。
