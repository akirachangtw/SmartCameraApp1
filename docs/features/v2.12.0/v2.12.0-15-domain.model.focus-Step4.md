package com.akirachang.ai.photographer.presentation.camera.focus

import android.graphics.RectF
import android.util.Log
import com.akirachang.ai.photographer.domain.model.focus.*

/**
 * Default implementation of FocusStateMachine.
 * v2.12.0-15 / §17.11.9
 *
 * Three orthogonal state axes (§6.2):
 *   1. VisualTargetState  — driven by TrackedTargetState + EyeAfDecision.visualState
 *   2. LensFocusState     — driven by FocusControllerResult.opticalResult
 *   3. ExposureLockState  — driven by UserIntentState.isAeAfLockRequested
 *
 * SystemEvent handling:
 *   LENS_SWITCHED / LIFECYCLE_STOPPED → always return FocusUiState.EMPTY
 */
class DefaultFocusStateMachine : FocusStateMachine {

    companion object {
        private const val TAG = "DefaultFocusStateMachine"
    }

    override fun reduce(input: FocusStateInput): FocusUiState {
        // ── System interrupt events — always reset to EMPTY ──────────────────
        if (input.systemEvent == SystemEvent.LENS_SWITCHED ||
            input.systemEvent == SystemEvent.LIFECYCLE_STOPPED) {
            Log.i(TAG, "[v2.12.0-15] SystemEvent=${input.systemEvent} → EMPTY reset")
            return FocusUiState.EMPTY
        }

        // ── Axis 1: VisualTargetState ─────────────────────────────────────────
        val visualState = reduceVisualState(input)

        // ── Axis 2: LensFocusState ────────────────────────────────────────────
        val lensState = reduceLensState(input)

        // ── Axis 3: ExposureLockState ─────────────────────────────────────────
        val exposureState = reduceExposureState(input)

        // ── OverlayModel: null when visual is None ────────────────────────────
        val overlay = if (visualState == VisualTargetState.None) {
            null
        } else {
            buildOverlayModel(input, visualState, exposureState)
        }

        return FocusUiState(
            visualTargetState = visualState,
            lensFocusState = lensState,
            exposureLockState = exposureState,
            overlayModel = overlay,
        )
    }

    // ── Axis 1: VisualTargetState reducer ─────────────────────────────────────
    private fun reduceVisualState(input: FocusStateInput): VisualTargetState {
        val primary = input.trackedTargetState.primaryTarget
        val decisionVisual = input.eyeAfDecision.visualState

        // No primary target → None (消滅殭屍框)
        if (primary == null) {
            val lost = input.trackedTargetState.recentlyLostTarget
            return if (lost != null) VisualTargetState.TrackingFade else VisualTargetState.None
        }

        // Delegate to EyeAfDecision visual state (policy layer owns this)
        return decisionVisual
    }

    // ── Axis 2: LensFocusState reducer ───────────────────────────────────────
    private fun reduceLensState(input: FocusStateInput): LensFocusState {
        val optical = input.focusControllerResult?.opticalResult
            ?: return LensFocusState.NotRequested

        return when (optical) {
            OpticalFocusState.FOCUSED            -> LensFocusState.Focused
            OpticalFocusState.NOT_FOCUSED        -> LensFocusState.NotFocused
            OpticalFocusState.PASSIVE_CONTINUOUS -> LensFocusState.PassiveContinuous
            OpticalFocusState.UNSUPPORTED        -> LensFocusState.UnsupportedFixedFocus
            OpticalFocusState.UNKNOWN            -> LensFocusState.NotRequested
        }
    }

    // ── Axis 3: ExposureLockState reducer ─────────────────────────────────────
    private fun reduceExposureState(input: FocusStateInput): ExposureLockState {
        return if (input.userIntent.isAeAfLockRequested) {
            ExposureLockState.AEAFLocked
        } else {
            ExposureLockState.Unlocked
        }
    }

    // ── OverlayModel builder ──────────────────────────────────────────────────
    private fun buildOverlayModel(
        input: FocusStateInput,
        visualState: VisualTargetState,
        exposureState: ExposureLockState,
    ): OverlayModel {
        val primary = input.trackedTargetState.primaryTarget
        val primaryBox = primary?.boundingBoxView

        val color = if (exposureState == ExposureLockState.AEAFLocked) {
            OverlayColor.YELLOW
        } else {
            OverlayColor.CYAN
        }

        val alpha = when (visualState) {
            VisualTargetState.VisualLocked   -> 1.0f
            VisualTargetState.TrackingStable -> 1.0f
            VisualTargetState.Candidate      -> 0.6f
            VisualTargetState.TrackingFade   -> 0.4f
            VisualTargetState.Lost           -> 0.2f
            VisualTargetState.None           -> 0.0f
        }

        val animation = when (visualState) {
            VisualTargetState.VisualLocked   -> OverlayAnimation.NONE
            VisualTargetState.TrackingFade   -> OverlayAnimation.FADE
            VisualTargetState.Candidate      -> OverlayAnimation.PULSE
            else                             -> OverlayAnimation.NONE
        }

        val showLockIcon = exposureState == ExposureLockState.AEAFLocked

        val secondaryBoxes = input.trackedTargetState.allTargets
            .filter { it != primary }
            .mapNotNull { it.boundingBoxView }

        return OverlayModel(
            primaryBox = primaryBox,
            secondaryBoxes = secondaryBoxes,
            primaryColor = color,
            alpha = alpha,
            showLockIcon = showLockIcon,
            animation = animation,
        )
    }
}

package com.akirachang.ai.photographer.presentation.camera.focus

import android.util.Log
import com.akirachang.ai.photographer.domain.model.focus.VisualTargetState
import kotlin.math.min
import kotlin.math.max
import kotlin.math.sqrt

/**
 * Focus trigger throttle controller.
 * v2.12.0-15 / §17.12.2
 *
 * Extracted from AfTriggerThrottleController with domain-layer decoupling:
 * - Accepts VisualTargetState instead of AfState
 * - Adds Target Lost Grace Period (500ms) to prevent 10s stall bug
 * - Removes direct AfState dependency
 *
 * NOTE: AfTriggerThrottleController in AfStateMachine.kt is kept for backward
 * compatibility with CameraViewModel. This class is for Step 5+ usage.
 */
class FocusTriggerThrottle(
    private val preLockThrottleMs:   Long  = 1_500L,
    private val postLockCooldownMs:  Long  = 5_000L,
    private val movingCooldownMs:    Long  = 2_000L,
    private val backoffBaseMs:       Long  = 2_000L,
    private val targetLostGraceMs:   Long  = 500L,    // v2.12.0-15: Grace period before forcing None
    var rescanThresholdPx:           Float = 54f,
) {
    companion object {
        private const val TAG = "FocusTriggerThrottle"
        private const val MAX_FAIL_COUNT = 3
    }

    private var lastTriggerTimeMs   = 0L
    private var nextAllowedTimeMs   = 0L
    private var isArmed             = false
    private var lockedTimestamp     = 0L
    private var consecutiveFailCount = 0
    private var targetLostTimeMs    = 0L   // v2.12.0-15: when target was last seen

    private val stabilityTracker = InternalStabilityTracker()

    fun setScreenWidth(widthPx: Int) {
        rescanThresholdPx = widthPx * 0.05f
    }

    fun onSuccess() {
        isArmed              = true
        lockedTimestamp      = now()
        nextAllowedTimeMs    = now() + postLockCooldownMs
        consecutiveFailCount = 0
        targetLostTimeMs     = 0L
        Log.d(TAG, "[v2.12.0-15] onSuccess: armed, cooldown=${postLockCooldownMs}ms")
    }

    fun onFailure(failCount: Int) {
        val backoff = min(backoffBaseMs * (1L shl (failCount - 1).coerceIn(0, 2)), 8_000L)
        nextAllowedTimeMs = now() + backoff
        consecutiveFailCount++
        if (consecutiveFailCount >= MAX_FAIL_COUNT) {
            Log.i(TAG, "[v2.12.0-15] Auto-retry suspended (failCount=$consecutiveFailCount)")
        }
    }

    /**
     * v2.12.0-15: Target Lost Grace Period.
     * Call when primary target is lost. After [targetLostGraceMs], shouldTrigger() returns false.
     */
    fun onTargetLost() {
        if (targetLostTimeMs == 0L) {
            targetLostTimeMs = now()
            Log.d(TAG, "[v2.12.0-15] Target lost, grace period started (${targetLostGraceMs}ms)")
        }
    }

    /**
     * Call when target is reacquired.
     */
    fun onTargetReacquired() {
        targetLostTimeMs = 0L
    }

    fun shouldTrigger(
        eyeScreenX: Float,
        eyeScreenY: Float,
        currentVisualState: VisualTargetState,
    ): Boolean {
        val n = now()

        // v2.12.0-15: Target Lost Grace Period — force stop after grace period
        if (targetLostTimeMs > 0L && (n - targetLostTimeMs) > targetLostGraceMs) {
            Log.d(TAG, "[v2.12.0-15] Grace period expired → shouldTrigger=false")
            return false
        }

        // Suspended mode
        if (consecutiveFailCount >= MAX_FAIL_COUNT) return false

        // Don't interrupt in-flight FOCUSING equivalent
        if (currentVisualState == VisualTargetState.Candidate) return false

        val effectiveThreshold = if (isArmed && currentVisualState == VisualTargetState.VisualLocked)
            rescanThresholdPx * 2.5f else rescanThresholdPx

        val deltaPx  = stabilityTracker.getDeltaPx(eyeScreenX, eyeScreenY)
        val isMoving = deltaPx > effectiveThreshold

        if (isArmed && isMoving && currentVisualState == VisualTargetState.VisualLocked) {
            val potentialMoveTime = lockedTimestamp + movingCooldownMs
            nextAllowedTimeMs = min(nextAllowedTimeMs, max(n, potentialMoveTime))
        }

        if (n < nextAllowedTimeMs) return false

        if (!isArmed && currentVisualState == VisualTargetState.None) {
            if (n - lastTriggerTimeMs < preLockThrottleMs) return false
        }

        lastTriggerTimeMs = n
        return true
    }

    fun reset() {
        lastTriggerTimeMs    = 0L
        nextAllowedTimeMs    = 0L
        isArmed              = false
        lockedTimestamp      = 0L
        consecutiveFailCount = 0
        targetLostTimeMs     = 0L
        stabilityTracker.reset()
        Log.d(TAG, "[v2.12.0-15] FocusTriggerThrottle reset")
    }

    private fun now() = System.currentTimeMillis()

    // Internal stability tracker (domain-decoupled copy of StabilityTracker)
    private class InternalStabilityTracker {
        private var lastX = Float.NaN
        private var lastY = Float.NaN

        fun getDeltaPx(x: Float, y: Float): Float {
            if (lastX.isNaN()) { lastX = x; lastY = y; return 0f }
            val dx = x - lastX; val dy = y - lastY
            lastX = x; lastY = y
            return sqrt(dx * dx + dy * dy)
        }

        fun reset() { lastX = Float.NaN; lastY = Float.NaN }
    }
}
